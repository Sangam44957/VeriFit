/**
 * OAuth Service - Google OAuth Login Flow
 * 
 * Responsibilities:
 * - Generate OAuth authorization URLs (redirect to Google)
 * - Handle OAuth callback (code → tokens)
 * - Exchange authorization code for Google access token
 * - Retrieve user profile from Google
 * - Manage OAuth state (CSRF protection)
 * - Create LoginResponse with JWT + user info
 * 
 * Security Considerations:
 * - State parameter prevents CSRF attacks
 * - State is single-use (deleted after verification)
 * - State expires after 10 minutes
 * - Never log access tokens or sensitive data
 * - Verify issuer (iss) matches Google
 * - All network calls have timeouts
 */

import { Injectable, Logger, BadRequestException } from '@nestjs/common';
import { randomUUID } from 'crypto';
import fetch from 'node-fetch';
import { z } from 'zod';
import { JwtService } from './jwt.service';
import {
  GoogleOAuthProfile,
  OAuthStateMetadata,
  LoginResponse,
  AuthenticatedUser,
} from '../types';

/**
 * Google OAuth Configuration
 * Typically injected from environment variables
 */
export interface GoogleOAuthConfig {
  /** OAuth Client ID (from Google Cloud Console) */
  clientId: string;

  /** OAuth Client Secret (from Google Cloud Console) */
  clientSecret: string;

  /** Redirect URI registered with Google (e.g., http://localhost:3001/api/v1/auth/callback) */
  redirectUri: string;
}

/**
 * OAuth State stored temporarily during auth flow
 * Prevents CSRF by validating state on callback
 */
interface OAuthState {
  state: string;
  expiresAt: Date;
  organizationId?: string;
  userId?: string;
}

/**
 * Google OAuth Token Response
 * Returned by Google token endpoint
 */
interface GoogleTokenResponse {
  access_token: string;
  expires_in: number;
  scope: string;
  token_type: 'Bearer';
  id_token?: string;
}

/**
 * Google OAuth UserInfo Response
 * Returned by Google userinfo endpoint
 */
interface GoogleUserInfoResponse {
  sub: string; // Google user ID
  name?: string;
  given_name?: string;
  family_name?: string;
  picture?: string;
  email: string;
  email_verified: boolean;
  locale?: string;
}

/**
 * Login Request Data
 * What frontend sends to initiate login
 */
const LoginInitiateRequestSchema = z.object({
  organizationId: z.string().optional(),
  redirectUri: z.string().url().optional(),
});

export type LoginInitiateRequest = z.infer<
  typeof LoginInitiateRequestSchema
>;

/**
 * Callback Request Data
 * What Google sends us back to
 */
const OAuthCallbackRequestSchema = z.object({
  code: z.string().min(1, 'Authorization code required'),
  state: z.string().min(1, 'State parameter required'),
});

export type OAuthCallbackRequest = z.infer<
  typeof OAuthCallbackRequestSchema
>;

/**
 * OAuth Service - Handles Google OAuth login flow
 *
 * @example
 * ```typescript
 * const oauthService = new OAuthService(jwtService, {
 *   clientId: process.env.OAUTH_CLIENT_ID,
 *   clientSecret: process.env.OAUTH_CLIENT_SECRET,
 *   redirectUri: process.env.OAUTH_REDIRECT_URI
 * });
 *
 * // Step 1: Generate login URL
 * const { authorizationUrl, state } = oauthService.generateAuthorizationUrl();
 * // Redirect user to authorizationUrl
 * // Store state in database with TTL
 *
 * // Step 2: Handle callback
 * const loginResponse = await oauthService.handleCallback({
 *   code: 'auth-code-from-google',
 *   state: 'state-from-database'
 * });
 * // Return JWT + user info to frontend
 * ```
 */
@Injectable()
export class OAuthService {
  private readonly logger = new Logger(OAuthService.name);
  private readonly config: GoogleOAuthConfig;
  private readonly jwtService: JwtService;

  // In-memory state store for development
  // Phase 6: migrate to Redis for scalability
  private oauthStates: Map<string, OAuthState> = new Map();

  // Google OAuth endpoints (constants)
  private readonly GOOGLE_OAUTH_URL =
    'https://accounts.google.com/o/oauth2/v2/auth';
  private readonly GOOGLE_TOKEN_URL = 'https://oauth2.googleapis.com/token';
  private readonly GOOGLE_USERINFO_URL =
    'https://openidconnect.googleapis.com/v1/userinfo';

  // OAuth scopes: email + profile
  private readonly OAUTH_SCOPE = [
    'openid',
    'email',
    'profile',
  ].join(' ');

  constructor(jwtService: JwtService, config: GoogleOAuthConfig) {
    this.validateConfig(config);
    this.jwtService = jwtService;
    this.config = config;

    this.logger.log('OAuth Service initialized for Google');
  }

  /**
   * Validate OAuth configuration at startup
   *
   * @throws Error if configuration is invalid
   */
  private validateConfig(config: GoogleOAuthConfig): void {
    if (!config.clientId || !config.clientSecret || !config.redirectUri) {
      throw new Error(
        'Google OAuth requires clientId, clientSecret, and redirectUri'
      );
    }

    try {
      new URL(config.redirectUri);
    } catch (error) {
      throw new Error(`Invalid redirectUri: ${config.redirectUri}`);
    }

    this.logger.log(
      `OAuth configured with redirectUri: ${config.redirectUri}`
    );
  }

  /**
   * Generate authorization URL for login
   *
   * Flow:
   * 1. Create random state (CSRF token)
   * 2. Build Google OAuth URL with state
   * 3. Return URL + state (caller will store state in DB)
   *
   * @param organizationId - Optional: organization for multi-org login (Phase 2)
   * @returns Authorization URL and state for database storage
   *
   * @example
   * ```typescript
   * const { authorizationUrl, state } = oauthService.generateAuthorizationUrl();
   * // Redirect user: res.redirect(authorizationUrl)
   * // Store state in DB: db.oauthState.create({ state, expiresAt: now + 10min })
   * ```
   */
  generateAuthorizationUrl(
    organizationId?: string
  ): { authorizationUrl: string; state: string; expiresAt: Date } {
    // Generate random state (CSRF token)
    const state = randomUUID();

    // Calculate expiration (10 minutes from now)
    const expiresAt = new Date(Date.now() + 10 * 60 * 1000);

    // Store state in memory (Phase 1)
    // Phase 6: migrate to Redis
    this.oauthStates.set(state, {
      state,
      expiresAt,
      organizationId,
    });

    // Clean up expired states periodically
    this.cleanupExpiredStates();

    // Build authorization URL
    const params = new URLSearchParams({
      client_id: this.config.clientId,
      redirect_uri: this.config.redirectUri,
      response_type: 'code',
      scope: this.OAUTH_SCOPE,
      state, // CSRF protection
      access_type: 'offline', // Request refresh token (for Phase 6)
      prompt: 'consent', // Always show consent screen
    });

    const authorizationUrl = `${this.GOOGLE_OAUTH_URL}?${params.toString()}`;

    this.logger.debug(`Generated authorization URL for state: ${state}`);

    return {
      authorizationUrl,
      state,
      expiresAt,
    };
  }

  /**
   * Verify OAuth state (CSRF protection)
   *
   * Flow:
   * 1. Lookup state in store
   * 2. Check it exists and not expired
   * 3. Delete state (single-use)
   * 4. Return state metadata or throw
   *
   * @param state - State from OAuth callback
   * @returns State metadata
   * @throws BadRequestException if state invalid or expired
   *
   * @example
   * ```typescript
   * try {
   *   const oauthState = oauthService.verifyState(stateFromCallback);
   *   // State is valid, proceed with code exchange
   * } catch (error) {
   *   // State invalid or expired → unauthorized
   * }
   * ```
   */
  verifyOAuthState(state: string): OAuthState {
    if (!state) {
      this.logger.warn('OAuth state verification failed: state is empty');
      throw new BadRequestException('State parameter required');
    }

    const oauthState = this.oauthStates.get(state);

    if (!oauthState) {
      this.logger.warn(
        `OAuth state verification failed: state not found. Possible CSRF attack.`
      );
      throw new BadRequestException('Invalid state parameter (CSRF protection)');
    }

    if (oauthState.expiresAt < new Date()) {
      this.oauthStates.delete(state);
      this.logger.warn(
        `OAuth state verification failed: state expired. Possible replay attack.`
      );
      throw new BadRequestException(
        'State parameter expired. Please restart login.'
      );
    }

    // Delete state (single-use)
    this.oauthStates.delete(state);

    this.logger.debug(`OAuth state verified and consumed: ${state}`);

    return oauthState;
  }

  /**
   * Exchange authorization code for access token
   *
   * Makes HTTP request to Google token endpoint.
   * Converts authorization code → access_token + id_token.
   *
   * @param code - Authorization code from Google callback
   * @returns Google token response
   * @throws Error if code exchange fails
   *
   * @example
   * ```typescript
   * const tokens = await oauthService.exchangeCodeForToken(authCode);
   * const profile = await oauthService.getUserProfile(tokens.access_token);
   * ```
   */
  private async exchangeCodeForToken(
    code: string
  ): Promise<GoogleTokenResponse> {
    try {
      const params = new URLSearchParams({
        client_id: this.config.clientId,
        client_secret: this.config.clientSecret,
        code,
        grant_type: 'authorization_code',
        redirect_uri: this.config.redirectUri,
      });

      const response = await fetch(this.GOOGLE_TOKEN_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: params.toString(),
        timeout: 10000, // 10 second timeout
      });

      if (!response.ok) {
        this.logger.error(
          `Google token exchange failed: ${response.status} ${response.statusText}`
        );
        throw new Error(
          `Google token exchange failed: ${response.statusText}`
        );
      }

      const data = (await response.json()) as GoogleTokenResponse;

      if (!data.access_token) {
        this.logger.error('Google token response missing access_token');
        throw new Error('Token response missing access_token');
      }

      this.logger.debug('Successfully exchanged code for access token');

      return data;
    } catch (error) {
      this.logger.error(`Failed to exchange authorization code: ${error}`);
      throw new Error('Failed to exchange authorization code');
    }
  }

  /**
   * Retrieve user profile from Google
   *
   * Makes HTTP request to Google userinfo endpoint.
   * Retrieves email, name, picture, etc.
   *
   * @param accessToken - Google access token
   * @returns Google user profile
   * @throws Error if profile retrieval fails
   *
   * @example
   * ```typescript
   * const profile = await oauthService.getUserProfile(accessToken);
   * console.log(profile.email); // User's Google email
   * ```
   */
  private async getUserProfile(
    accessToken: string
  ): Promise<GoogleUserInfoResponse> {
    try {
      const response = await fetch(this.GOOGLE_USERINFO_URL, {
        method: 'GET',
        headers: {
          Authorization: `Bearer ${accessToken}`,
          'Accept': 'application/json',
        },
        timeout: 10000, // 10 second timeout
      });

      if (!response.ok) {
        this.logger.error(
          `Google userinfo failed: ${response.status} ${response.statusText}`
        );
        throw new Error(
          `Google userinfo failed: ${response.statusText}`
        );
      }

      const profile = (await response.json()) as GoogleUserInfoResponse;

      if (!profile.email) {
        this.logger.error('Google profile missing email');
        throw new Error('Profile missing required email field');
      }

      this.logger.debug(`Retrieved profile for Google user: ${profile.sub}`);

      return profile;
    } catch (error) {
      this.logger.error(`Failed to retrieve user profile: ${error}`);
      throw new Error('Failed to retrieve user profile');
    }
  }

  /**
   * Convert Google profile to VeriFit user format
   *
   * Maps Google userinfo → AuthenticatedUser
   *
   * @param googleProfile - Google user profile
   * @param role - User role (from database pre-creation)
   * @param organizationId - Organization ID (from database)
   * @returns VeriFit AuthenticatedUser
   */
  private mapGoogleProfileToUser(
    googleProfile: GoogleUserInfoResponse,
    role: 'ADMIN' | 'STAFF' | 'STUDENT',
    organizationId: string,
    userId: string
  ): AuthenticatedUser {
    return {
      id: userId, // Database user ID
      email: googleProfile.email,
      organizationId,
      role,
      lastLoginAt: new Date(),
      accountStatus: 'ACTIVE',
    };
  }

  /**
   * Handle OAuth callback
   *
   * Complete OAuth flow:
   * 1. Verify state (CSRF protection)
   * 2. Exchange code for access token
   * 3. Retrieve user profile
   * 4. Lookup user in database (pre-created)
   * 5. Generate JWT
   * 6. Create login response
   *
   * @param request - Callback request (code + state from Google)
   * @param getUserFromDatabase - Function to lookup user by email
   * @returns LoginResponse (JWT + user info)
   * @throws BadRequestException if any step fails
   *
   * @example
   * ```typescript
   * const loginResponse = await oauthService.handleCallback(
   *   { code: '...', state: '...' },
   *   async (email) => {
   *     const user = await db.user.findUnique({ where: { email } });
   *     return user;
   *   }
   * );
   * // Response contains JWT + user info
   * ```
   */
  async handleCallback(
    request: OAuthCallbackRequest,
    getUserFromDatabase: (email: string) => Promise<{
      id: string;
      email: string;
      role: 'ADMIN' | 'STAFF' | 'STUDENT';
      organizationId: string;
      accountStatus: 'ACTIVE' | 'LOCKED' | 'SUSPENDED';
    } | null>
  ): Promise<LoginResponse> {
    // Validate input
    try {
      OAuthCallbackRequestSchema.parse(request);
    } catch (error) {
      this.logger.warn(`OAuth callback validation failed: ${error}`);
      throw new BadRequestException('Invalid callback parameters');
    }

    // Step 1: Verify state (CSRF protection)
    const oauthState = this.verifyOAuthState(request.state);

    try {
      // Step 2: Exchange code for access token
      const tokens = await this.exchangeCodeForToken(request.code);

      // Step 3: Retrieve user profile from Google
      const googleProfile = await this.getUserProfile(tokens.access_token);

      // Step 4: Lookup user in database (pre-created by admin)
      const user = await getUserFromDatabase(googleProfile.email);

      if (!user) {
        this.logger.warn(
          `User not found for email: ${googleProfile.email}. User must be pre-created.`
        );
        throw new BadRequestException(
          'User not found. Please contact administrator.'
        );
      }

      // Check account status
      if (user.accountStatus === 'LOCKED') {
        this.logger.warn(`Login attempted on locked account: ${user.email}`);
        throw new BadRequestException(
          'Account is locked. Please contact administrator.'
        );
      }

      if (user.accountStatus === 'SUSPENDED') {
        this.logger.warn(
          `Login attempted on suspended account: ${user.email}`
        );
        throw new BadRequestException('Account suspended.');
      }

      // Step 5: Generate JWT
      const jwtToken = this.jwtService.generateToken({
        userId: user.id,
        email: user.email,
        organizationId: user.organizationId,
        role: user.role,
      });

      // Step 6: Map to authenticated user
      const authenticatedUser = this.mapGoogleProfileToUser(
        googleProfile,
        user.role,
        user.organizationId,
        user.id
      );

      // Get token expiration
      const expirationDate = this.jwtService.getExpirationDate(jwtToken);
      if (!expirationDate) {
        throw new Error('Failed to determine token expiration');
      }

      const loginResponse: LoginResponse = {
        token: jwtToken,
        user: authenticatedUser,
        expiresAt: Math.floor(expirationDate.getTime() / 1000), // unix timestamp
        redirectUrl: '/dashboard',
      };

      this.logger.log(`Successful login for user: ${user.email}`);

      return loginResponse;
    } catch (error) {
      // Don't re-throw BadRequestException (already thrown above)
      if (error instanceof BadRequestException) {
        throw error;
      }

      // Log other errors without exposing details to client
      this.logger.error(`OAuth callback failed: ${error}`);

      throw new BadRequestException(
        'Login failed. Please try again or contact support.'
      );
    }
  }

  /**
   * Clean up expired states periodically
   *
   * Prevents memory leaks from accumulating old states
   * Called after every generateAuthorizationUrl
   *
   * @private
   */
  private cleanupExpiredStates(): void {
    const now = new Date();
    let deleted = 0;

    for (const [state, oauthState] of this.oauthStates.entries()) {
      if (oauthState.expiresAt < now) {
        this.oauthStates.delete(state);
        deleted++;
      }
    }

    if (deleted > 0) {
      this.logger.debug(`Cleaned up ${deleted} expired OAuth states`);
    }
  }

  /**
   * Get current number of active OAuth states
   * Use for monitoring/debugging only
   *
   * @returns Number of active states in memory
   */
  getActiveStatesCount(): number {
    this.cleanupExpiredStates();
    return this.oauthStates.size;
  }
}
