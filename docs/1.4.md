  USER = 'USER',
  STUDENT = 'STUDENT',
  ORGANIZATION = 'ORGANIZATION',
  PLACEMENT_DRIVE = 'PLACEMENT_DRIVE',

/**
 * Permission - defines what a role can do
 * Maps role → actions they can perform
 */
export interface Permission {
  /** Role that has this permission */
  role: UserRole;

  /** Resource type being accessed */
  resource: ResourceType;

  /** Actions allowed on this resource */
  actions: ResourceAction[];

  /** Whether this permission applies across all orgs (ADMIN only) */
  isGlobal: boolean;

  /** Whether this permission requires org scoping check */
  requiresOrgScoping: boolean;
}

/**
 * Authorization Result - detailed response from auth checks
 */
export interface AuthorizationResult {
  /** Is access allowed? */
  allowed: boolean;

  /** Reason for denial (if not allowed) */
  reason?: string;

  /** Which role denied access (for audit) */
  deniedBy?: UserRole;

  /** Resource being accessed */
  resource: ResourceType;

  /** Action being attempted */
  action: ResourceAction;

  /** Timestamp of check */
  checkedAt: Date;
}

/**
 * Access Context - all data needed for authorization decision
 */
export interface AccessContext {
  /** User attempting access */
  user: AuthenticatedUser;

  /** Resource type being accessed */
  resource: ResourceType;

  /** Action being attempted */
  action: ResourceAction;

  /** ID of resource being accessed (e.g., userId, studentId) */
  resourceId?: string;

  /** Organization ID of resource (for org scoping) */
  resourceOrgId?: string;

  /** Additional context (e.g., query params, path) */
  metadata?: Record<string, any>;
}

/**
 * Authorization Service - Role-Based Access Control (RBAC)
 *
 * Responsibilities:
 * - Define what each role can do (permissions matrix)
 * - Check if user can perform action on resource
 * - Enforce organization scoping (user can only access own org)
 * - Prevent IDOR (can't access other user's data by ID manipulation)
 * - Provide audit trail (log all authorization checks)
 *
 * Security Invariants Enforced:
 * - INVARIANT 003: Eligibility ≠ Score (not auth's job, but similar pattern)
 * - Default deny, explicit allow
 * - Organization isolation (no cross-org leakage)
 * - Role hierarchy: ADMIN > STAFF > STUDENT
 *
 * IDOR Prevention:
 * - Every resource access checks organizationId
 * - User can only access resources in their org
 * - Except ADMIN who can access all orgs
 * - Attempt to access other org's resource → 403 Forbidden
 */

import { Injectable, Logger, ForbiddenException } from '@nestjs/common';
import {
  AuthenticatedUser,
  ResourceAction,
  ResourceType,
  UserRole,
  Permission,
  AuthorizationResult,
  AccessContext,
} from '../types';

/**
 * Authorization Service
 *
 * @example
 * ```typescript
 * const authzService = new AuthorizationService();
 *
 * // Check if user can list students in their org
 * const result = authzService.authorize({
 *   user: { id: 'user-123', role: 'STAFF', organizationId: 'org-456' },
 *   resource: ResourceType.STUDENT,
 *   action: ResourceAction.LIST_STUDENTS,
 *   resourceOrgId: 'org-456' // user's org
 * });
 *
 * if (!result.allowed) {
 *   throw new ForbiddenException(result.reason);
 * }
 *
 * // Proceed with listing students
 * ```
 */
@Injectable()
export class AuthorizationService {
  private readonly logger = new Logger(AuthorizationService.name);

  /**
   * Permission Matrix - defines what each role can do
   *
   * Structure:
   * - ADMIN: can do everything (global access)
   * - STAFF: can manage org and view students
   * - STUDENT: can read own profile only
   *
   * Phase 2+: Can be extended with more granular permissions
   */
  private readonly permissionMatrix: Permission[] = [
    // ============================================
    // ADMIN - Full Access
    // ============================================
    {
      role: 'ADMIN',
      resource: ResourceType.USER,
      actions: [
        ResourceAction.CREATE_USER,
        ResourceAction.READ_USER,
        ResourceAction.UPDATE_USER,
        ResourceAction.DELETE_USER,
        ResourceAction.LIST_USERS,
      ],
      isGlobal: true, // ADMIN accesses all orgs
      requiresOrgScoping: false,
    },
    {
      role: 'ADMIN',
      resource: ResourceType.STUDENT,
      actions: [
        ResourceAction.CREATE_STUDENT,
        ResourceAction.READ_STUDENT,
        ResourceAction.UPDATE_STUDENT,
        ResourceAction.DELETE_STUDENT,
        ResourceAction.LIST_STUDENTS,
      ],
      isGlobal: true,
      requiresOrgScoping: false,
    },
    {
      role: 'ADMIN',
      resource: ResourceType.ORGANIZATION,
      actions: [
        ResourceAction.READ_ORGANIZATION,
        ResourceAction.UPDATE_ORGANIZATION,
        ResourceAction.MANAGE_ORGANIZATION,
      ],
      isGlobal: true,
      requiresOrgScoping: false,
    },

    // ============================================
    // STAFF - Manage Own Organization
    // ============================================
    {
      role: 'STAFF',
      resource: ResourceType.USER,
      actions: [
        ResourceAction.READ_USER,
        ResourceAction.LIST_USERS,
        // STAFF cannot create/update/delete users (ADMIN only)
      ],
      isGlobal: false,
      requiresOrgScoping: true, // STAFF scoped to own org
    },
    {
      role: 'STAFF',
      resource: ResourceType.STUDENT,
      actions: [
        ResourceAction.CREATE_STUDENT,
        ResourceAction.READ_STUDENT,
        ResourceAction.UPDATE_STUDENT,
        ResourceAction.LIST_STUDENTS,
        // STAFF cannot delete students
      ],
      isGlobal: false,
      requiresOrgScoping: true,
    },
    {
      role: 'STAFF',
      resource: ResourceType.ORGANIZATION,
      actions: [
        ResourceAction.READ_ORGANIZATION,
        // STAFF cannot manage org settings
      ],
      isGlobal: false,
      requiresOrgScoping: true,
    },

    // ============================================
    // STUDENT - Read Own Data Only
    // ============================================
    {
      role: 'STUDENT',
      resource: ResourceType.USER,
      actions: [
        ResourceAction.READ_USER, // Can read own profile only
      ],
      isGlobal: false,
      requiresOrgScoping: true,
    },
    {
      role: 'STUDENT',
      resource: ResourceType.STUDENT,
      actions: [
        ResourceAction.READ_STUDENT, // Can read own student profile only
      ],
      isGlobal: false,
      requiresOrgScoping: true,
    },
    {
      role: 'STUDENT',
      resource: ResourceType.ORGANIZATION,
      actions: [
        ResourceAction.READ_ORGANIZATION, // Can see org they're in
      ],
      isGlobal: false,
      requiresOrgScoping: true,
    },
  ];

  constructor() {
    this.logger.log('Authorization Service initialized');
    this.logPermissionMatrix();
  }

  /**
   * Log permission matrix on startup (for debugging)
   * Shows what each role can do
   *
   * @private
   */
  private logPermissionMatrix(): void {
    const matrix = this.permissionMatrix.reduce(
      (acc, perm) => {
        if (!acc[perm.role]) {
          acc[perm.role] = [];
        }
        acc[perm.role].push(
          `${perm.resource}: [${perm.actions.join(', ')}]`
        );
        return acc;
      },
      {} as Record<string, string[]>
    );

    Object.entries(matrix).forEach(([role, resources]) => {
      this.logger.debug(`${role} permissions: ${resources.join(' | ')}`);
    });
  }

  /**
   * Main authorization check
   *
   * Flow:
   * 1. Get permissions for user's role
   * 2. Check if role can perform action on resource
   * 3. Check organization scoping (if required)
   * 4. Return allow/deny with reason
   *
   * @param context - Access context (user, resource, action, etc.)
   * @returns Authorization result (allowed + reason)
   *
   * @example
   * ```typescript
   * const result = authzService.authorize({
   *   user: authenticatedUser,
   *   resource: ResourceType.STUDENT,
   *   action: ResourceAction.LIST_STUDENTS,
   *   resourceOrgId: 'org-456'
   * });
   *
   * if (!result.allowed) {
   *   throw new ForbiddenException(result.reason);
   * }
   * ```
   */
  authorize(context: AccessContext): AuthorizationResult {
    const { user, resource, action, resourceOrgId, resourceId, metadata } =
      context;

    const result: AuthorizationResult = {
      allowed: false,
      resource,
      action,
      checkedAt: new Date(),
    };

    // Step 1: Get permissions for user's role
    const rolePermissions = this.permissionMatrix.filter(
      (p) => p.role === user.role && p.resource === resource
    );

    if (rolePermissions.length === 0) {
      result.reason = `No permissions defined for ${user.role} on ${resource}`;
      result.deniedBy = user.role;
      this.logger.warn(
        `Authorization denied: No permissions. User: ${user.id}, Role: ${user.role}, Resource: ${resource}, Action: ${action}`
      );
      return result;
    }

    // Step 2: Check if role can perform this action
    const permission = rolePermissions[0]; // Should only be one permission per role+resource
    const canPerformAction = permission.actions.includes(action);

    if (!canPerformAction) {
      result.reason = `${user.role} cannot perform ${action} on ${resource}`;
      result.deniedBy = user.role;
      this.logger.warn(
        `Authorization denied: Action not allowed. User: ${user.id}, Role: ${user.role}, Action: ${action}`
      );
      return result;
    }

    // Step 3: Check organization scoping (if required)
    if (permission.requiresOrgScoping) {
      const orgCheckResult = this.checkOrganizationScoping(
        user,
        resourceOrgId,
        resource,
        action
      );

      if (!orgCheckResult.allowed) {
        result.reason = orgCheckResult.reason;
        result.deniedBy = user.role;
        this.logger.warn(
          `Authorization denied: Org scoping failed. User: ${user.id}, User Org: ${user.organizationId}, Resource Org: ${resourceOrgId}`
        );
        return result;
      }
    }

    // Step 4: Check resource-level authorization (e.g., can STUDENT only read own profile?)
    const resourceCheckResult = this.checkResourceLevelAccess(
      user,
      resource,
      action,
      resourceId
    );

    if (!resourceCheckResult.allowed) {
      result.reason = resourceCheckResult.reason;
      result.deniedBy = user.role;
      this.logger.warn(
        `Authorization denied: Resource access check failed. User: ${user.id}, Resource: ${resource}, ResourceId: ${resourceId}`
      );
      return result;
    }

    // All checks passed
    result.allowed = true;
    this.logger.debug(
      `Authorization allowed. User: ${user.id}, Role: ${user.role}, Action: ${action} on ${resource}`
    );
    return result;
  }

  /**
   * Check organization scoping
   *
   * Rules:
   * - ADMIN can access any org (isGlobal: true)
   * - STAFF/STUDENT can only access own org
   * - If resourceOrgId doesn't match user's org, deny (IDOR prevention)
   *
   * @param user - Authenticated user
   * @param resourceOrgId - Organization of resource being accessed
   * @param resource - Resource type
   * @param action - Action being attempted
   * @returns Check result
   *
   * @private
   */
  private checkOrganizationScoping(
    user: AuthenticatedUser,
    resourceOrgId: string | undefined,
    resource: ResourceType,
    action: ResourceAction
  ): { allowed: boolean; reason?: string } {
    // ADMIN can access any org
    if (user.role === 'ADMIN') {
      return { allowed: true };
    }

    // STAFF/STUDENT must access own org only
    if (!resourceOrgId) {
      return {
        allowed: false,
        reason: 'Resource organization ID required for org scoping check',
      };
    }

    // IDOR Prevention: User's org must match resource's org
    if (user.organizationId !== resourceOrgId) {
      return {
        allowed: false,
        reason: `User cannot access resources in other organizations (IDOR prevention)`,
      };
    }

    return { allowed: true };
  }

  /**
   * Check resource-level access (fine-grained)
   *
   * Rules:
   * - STUDENT can only READ own profile (resourceId must match user.id)
   * - STUDENT cannot read other student's profile
   * - STAFF can read any student in their org
   * - ADMIN can read any student globally
   *
   * @param user - Authenticated user
   * @param resource - Resource type
   * @param action - Action being attempted
   * @param resourceId - ID of specific resource (e.g., userId, studentId)
   * @returns Check result
   *
   * @private
   */
  private checkResourceLevelAccess(
    user: AuthenticatedUser,
    resource: ResourceType,
    action: ResourceAction,
    resourceId?: string
  ): { allowed: boolean; reason?: string } {
    // ADMIN can access any resource
    if (user.role === 'ADMIN') {
      return { allowed: true };
    }

    // STUDENT trying to READ must be reading own profile
    if (user.role === 'STUDENT' && action === ResourceAction.READ_USER) {
      if (!resourceId) {
        // LIST_USERS not allowed for STUDENT
        if (action === ResourceAction.LIST_USERS) {
          return {
            allowed: false,
            reason: 'STUDENT cannot list all users',
          };
        }
      }

      // STUDENT can only read own user profile
      if (resourceId && resourceId !== user.id) {
        return {
          allowed: false,
          reason: 'STUDENT can only read own user profile (IDOR prevention)',
        };
      }
    }

    // STUDENT trying to READ STUDENT must be reading own profile
    if (user.role === 'STUDENT' && action === ResourceAction.READ_STUDENT) {
      if (resourceId && resourceId !== user.id) {
        return {
          allowed: false,
          reason: 'STUDENT can only read own student profile (IDOR prevention)',
        };
      }
    }

    // STUDENT trying to LIST_STUDENTS not allowed
    if (user.role === 'STUDENT' && action === ResourceAction.LIST_STUDENTS) {
      return {
        allowed: false,
        reason: 'STUDENT cannot list all students',
      };
    }

    return { allowed: true };
  }

  /**
   * Verify user can access a specific resource by ID
   *
   * Convenience method for common pattern:
   * "Can user X access resource Y (by ID) in org Z?"
   *
   * @param user - Authenticated user
   * @param resourceId - ID of resource to access
   * @param resourceOrgId - Organization of resource
   * @param action - Action being attempted
   * @param resource - Resource type
   * @returns true if allowed, false otherwise
   *
   * @example
   * ```typescript
   * // Can user access student record?
   * const canAccess = authzService.canAccessResource(
   *   user,
   *   'student-123',
   *   'org-456',
   *   ResourceAction.READ_STUDENT,
   *   ResourceType.STUDENT
   * );
   * ```
   */
  canAccessResource(
    user: AuthenticatedUser,
    resourceId: string,
    resourceOrgId: string,
    action: ResourceAction,
    resource: ResourceType
  ): boolean {
    const result = this.authorize({
      user,
      resource,
      action,
      resourceId,
      resourceOrgId,
    });

    return result.allowed;
  }

  /**
   * Get all permissions for a role
   *
   * Use for UI/UX: show user what they can do
   *
   * @param role - User role
   * @returns All permissions for this role
   *
   * @example
   * ```typescript
   * const staffPermissions = authzService.getPermissionsForRole('STAFF');
   * // Returns: [{ role: 'STAFF', resource: USER, actions: [...] }, ...]
   * ```
   */
  getPermissionsForRole(role: UserRole): Permission[] {
    return this.permissionMatrix.filter((p) => p.role === role);
  }

  /**
   * Get actions a role can perform on a resource
   *
   * @param role - User role
   * @param resource - Resource type
   * @returns Array of allowed actions, or empty if none
   *
   * @example
   * ```typescript
   * const studentCanDo = authzService.getAllowedActions('STUDENT', ResourceType.USER);
   * // Returns: [ResourceAction.READ_USER]
   * ```
   */
  getAllowedActions(role: UserRole, resource: ResourceType): ResourceAction[] {
    const permission = this.permissionMatrix.find(
      (p) => p.role === role && p.resource === resource
    );

    return permission?.actions ?? [];
  }

  /**
   * Check if role is global (ADMIN) or org-scoped
   *
   * @param role - User role
   * @returns true if role has global access, false if org-scoped
   */
  isGlobalRole(role: UserRole): boolean {
    const hasGlobal = this.permissionMatrix.some(
      (p) => p.role === role && p.isGlobal
    );

    return hasGlobal;
  }

  /**
   * Throw ForbiddenException if not authorized
   *
   * Helper for use in controllers/guards
   *
   * @param context - Access context
   * @throws ForbiddenException if not authorized
   *
   * @example
   * ```typescript
   * // In controller
   * @Get('students/:id')
   * async getStudent(@Param('id') studentId: string, @Request() req) {
   *   this.authzService.assertAuthorized({
   *     user: req.user,
   *     resource: ResourceType.STUDENT,
   *     action: ResourceAction.READ_STUDENT,
   *     resourceId: studentId,
   *     resourceOrgId: req.user.organizationId
   *   });
   *
   *   // If we reach here, user is authorized
   *   return this.studentService.findById(studentId);
   * }
   * ```
   */
  assertAuthorized(context: AccessContext): void {
    const result = this.authorize(context);

    if (!result.allowed) {
      throw new ForbiddenException(
        result.reason || 'Access denied'
      );
    }
  }
}

What to KEEP ✅
Keep these core ideas:

1. Default deny, explicit allow.
This is exactly right for VeriFit authorization. 


2. Explicit permission matrix.
Using role + resource + action is a good foundation.

3. Three roles remain canonical:

ADMIN
STAFF
STUDENT
Do not introduce TPO again. STAFF remains the product mapping.

4. Organization isolation.
STAFF/STUDENT must never cross tenant boundaries.

5. Student self-access rules.
The idea that a student cannot simply change an ID in the URL and read another student's information is correct.

6. Separate authorization result from throwing HTTP exceptions.
The underlying authorization decision should ideally remain framework-independent.

What MUST be changed ❌
1. Biggest issue: resourceOrgId cannot be trusted from the request
The current model says:

resourceOrgId?: string;
and authorization compares it against the user's organization.

That's only safe if the resource's organization ID was obtained from a trusted server-side source.

The dangerous pattern is essentially:

resourceOrgId: req.body.organizationId
or:

resourceOrgId: req.query.organizationId
An attacker could simply submit their own organization ID while requesting another organization's resource.

Your example actually shows:

resourceOrgId: req.user.organizationId
for a student request. 


That does not prove that student-123 belongs to that organization.

Correct rule
For resource-by-ID authorization:

request resourceId
        ↓
trusted resource lookup
        ↓
actual resource.organizationId
        ↓
AuthorizationService
So 1.4 must explicitly say:

resourceOrgId MUST come from trusted server-side resource data, never from an untrusted client-supplied organization identifier.

This is the most important correction.

2. Student ID may not equal User ID ❌
The code currently does:

if (resourceId !== user.id)
for ResourceType.STUDENT. 


But our database architecture has separate User and Student concepts.

A Student resource may have:

student.id
student.userId
student.organizationId
Those are not necessarily the same identifier.

Therefore:

STUDENT resourceId === user.id
must not be assumed.

Correct pattern
For a student resource, ownership should eventually be established through:

student.userId === authenticatedUser.id
or through trusted resource ownership information supplied by the application layer.

Do not hardwire resourceId === user.id.

3. metadata: Record<string, any> violates our rules ❌
The document has:

metadata?: Record<string, any>;
We explicitly established a no-any baseline.

Change it to:

metadata?: Record<string, unknown>;
But honestly, I would avoid putting arbitrary metadata into the core authorization contract unless a real use case requires it.

4. Do NOT use @nestjs/common inside the core authorization service ❌
The document imports:

import {
  Injectable,
  Logger,
  ForbiddenException,
} from '@nestjs/common';


This is a boundary problem.

The authorization domain decision should not need NestJS.

Especially:

assertAuthorized()
should not be responsible for producing an HTTP framework exception.

Better:

AuthorizationService
    ↓
AuthorizationResult
Then later:

Nest AuthGuard
    ↓
AuthorizationService
    ↓
ForbiddenException
That fits the architecture much better.

Therefore
Remove from 1.4:

@Injectable()
Logger
ForbiddenException
from the core service.

Nest-specific translation belongs in 1.5 Guards & Middleware.

5. Don't claim "ADMIN > STAFF > STUDENT" as a hierarchy ❌
The document says:

Role hierarchy: ADMIN > STAFF > STUDENT


But the implementation is actually using explicit permissions, not inherited hierarchical permissions.

For example, STAFF doesn't automatically inherit everything STUDENT can do because of hierarchy; the matrix explicitly defines STAFF permissions.

So say:

ADMIN: broader/global permissions
STAFF: organization-scoped operational permissions
STUDENT: self-scoped permissions
Do not implement a generic "role hierarchy" mechanism yet.

6. "ADMIN can do everything" is too broad ❌
The service comment says:

ADMIN: can do everything (global access)



But the matrix doesn't actually define everything.

There are:

PLACEMENT_DRIVE
EVIDENCE
RANKING
resource types already declared, but no corresponding permissions for them. 


That creates a conceptual contradiction.

Correct rule
ADMIN is not "bypass authorization."

Instead:

ADMIN has explicitly defined global permissions.
Unknown resource/action → DENY.
This preserves our default-deny invariant.

7. Don't add future resources/actions prematurely ⚠️
The document introduces:

PLACEMENT_DRIVE
EVIDENCE
RANKING
and placement-drive actions even though those domains are explicitly Phase 2+. 


This is unnecessary design surface in 1.4.

I'd keep 1.4 focused on:

USER
STUDENT
ORGANIZATION
Then add Evidence/Ranking/Drive authorization when those domains actually exist.

This avoids inventing policies before we know the resource ownership models.

8. Permission arrays are accidentally mutable ❌
This is subtle but important.

The service has:

private readonly permissionMatrix: Permission[] = [...]
Then:

getPermissionsForRole()
returns the actual permission objects from that array. 


Likewise:

getAllowedActions()
returns the actual actions array. 


A caller could potentially mutate the returned array and alter authorization behavior.

Example conceptually:

authz.getAllowedActions('STAFF', STUDENT).push(DELETE_STUDENT);
That must never change the actual authorization policy.

Required
Return immutable copies, or store the matrix using immutable structures.

This should be explicitly tested.

9. The "audit trail" claim is misleading ⚠️
The service says:

But logging is not an audit trail.

Our architecture already has an audit package.

Authorization decisions should not pretend that application logs equal durable audit records.

For 1.4
Keep optional structured diagnostic logging if useful, but say:

Audit integration is handled through the audit layer.
Also avoid logging sensitive identifiers unnecessarily.
