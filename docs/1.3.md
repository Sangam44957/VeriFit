/**

- OAuth Service - Google OAuth Login Flow
-
- Responsibilities:
- - Generate OAuth authorization URLs (redirect to Google)
- - Handle OAuth callback (code → tokens)
- - Exchange authorization code for Google access token
- - Retrieve user profile from Google
- - Manage OAuth state (CSRF protection)
- - Create LoginResponse with JWT + user info
-
- Security Considerations:
- - State parameter prevents CSRF attacks
- - State is single-use (deleted after verification)
- - State expires after 10 minutes
- - Never log access tokens or sensitive data
- - Verify issuer (iss) matches Google
- - All network calls have timeouts
    */

@Injectable()
export class OAuthService {
private readonly logger = new Logger(OAuthService.name);
private readonly config: GoogleOAuthConfig;
private readonly jwtService: JwtService;

// In-memory state store for development
// Phase 6: migrate to Redis for scalability
private oauthStates: Map<string, OAuthState> = new Map();

// Google OAuth endpoints (constants)
private readonly GOOGLE_OAUTH_URL =
'https://accounts.google.com/o/oauth2/v2/auth';
private readonly GOOGLE_TOKEN_URL = 'https://oauth2.googleapis.com/token';
private readonly GOOGLE_USERINFO_URL =
'https://openidconnect.googleapis.com/v1/userinfo';

// OAuth scopes: email + profile
private readonly OAUTH_SCOPE = [
'openid',
'email',
'profile',
].join(' ');

constructor(jwtService: JwtService, config: GoogleOAuthConfig) {
this.validateConfig(config);
this.jwtService = jwtService;
this.config = config;

    this.logger.log('OAuth Service initialized for Google');

}

/**

-    Validate OAuth configuration at startup
-
-    @throws Error if configuration is invalid
     */
     private validateConfig(config: GoogleOAuthConfig): void {
     if (!config.clientId || !config.clientSecret || !config.redirectUri) {
     throw new Error(
     'Google OAuth requires clientId, clientSecret, and redirectUri'
     );
     }

    try {

      new URL(config.redirectUri);
    } catch (error) {
      throw new Error(`Invalid redirectUri: ${config.redirectUri}`);
    }

    this.logger.log(
      `OAuth configured with redirectUri: ${config.redirectUri}`
    );

}

/**

- Generate authorization URL for login
-
- Flow:
- 1.  Create random state (CSRF token)
- 2.  Build Google OAuth URL with state
- 3.  Return URL + state (caller will store state in DB)
-
- @param organizationId - Optional: organization for multi-org login (Phase 2)
- @returns Authorization URL and state for database storage
-
- @example
- ```typescript

  ```
- const { authorizationUrl, state } = oauthService.generateAuthorizationUrl();
- // Redirect user: res.redirect(authorizationUrl)
- // Store state in DB: db.oauthState.create({ state, expiresAt: now + 10min })
- ```

  ```

*/
generateAuthorizationUrl(
organizationId?: string
): { authorizationUrl: string; state: string; expiresAt: Date } {
// Generate random state (CSRF token)
const state = randomUUID();

    // Calculate expiration (10 minutes from now)
    const expiresAt = new Date(Date.now() + 10 * 60 * 1000);

    // Store state in memory (Phase 1)
    // Phase 6: migrate to Redis
    this.oauthStates.set(state, {
      state,
      expiresAt,
      organizationId,
    });

    // Clean up expired states periodically
    this.cleanupExpiredStates();

    // Build authorization URL
    const params = new URLSearchParams({
      client_id: this.config.clientId,
      redirect_uri: this.config.redirectUri,
      response_type: 'code',
      scope: this.OAUTH_SCOPE,
      state, // CSRF protection
      access_type: 'offline', // Request refresh token (for Phase 6)
      prompt: 'consent', // Always show consent screen
    });

    const authorizationUrl = `${this.GOOGLE_OAUTH_URL}?${params.toString()}`;

    this.logger.debug(`Generated authorization URL for state: ${state}`);

    return {
      authorizationUrl,
      state,
      expiresAt,
    };

}

/**

- Verify OAuth state (CSRF protection)
-
- Flow:
- 1.  Lookup state in store
- 2.  Check it exists and not expired
- 3.  Delete state (single-use)
- 4.  Return state metadata or throw
-
- @param state - State from OAuth callback
- @returns State metadata
- @throws BadRequestException if state invalid or expired
-
- @example
- ```typescript

  ```
- try {
- const oauthState = oauthService.verifyState(stateFromCallback);
- // State is valid, proceed with code exchange
- } catch (error) {
- // State invalid or expired → unauthorized
- }
- ```

  ```

*/
verifyOAuthState(state: string): OAuthState {
if (!state) {
this.logger.warn('OAuth state verification failed: state is empty');
throw new BadRequestException('State parameter required');
}

    const oauthState = this.oauthStates.get(state);

    if (!oauthState) {
      this.logger.warn(
        `OAuth state verification failed: state not found. Possible CSRF attack.`
      );
      throw new BadRequestException('Invalid state parameter (CSRF protection)');
    }

    if (oauthState.expiresAt < new Date()) {
      this.oauthStates.delete(state);
      this.logger.warn(
        `OAuth state verification failed: state expired. Possible replay attack.`
      );
      throw new BadRequestException(
        'State parameter expired. Please restart login.'
      );
    }

    // Delete state (single-use)
    this.oauthStates.delete(state);

    this.logger.debug(`OAuth state verified and consumed: ${state}`);

    return oauthState;

}

/**

- Exchange authorization code for access token
-
- Makes HTTP request to Google token endpoint.
- Converts authorization code → access_token + id_token.
-
- @param code - Authorization code from Google callback
- @returns Google token response
- @throws Error if code exchange fails
-
- @example
- ```typescript

  ```
- const tokens = await oauthService.exchangeCodeForToken(authCode);
- const profile = await oauthService.getUserProfile(tokens.access_token);
- ```

  ```

*/
private async exchangeCodeForToken(
code: string
): Promise<GoogleTokenResponse> {
try {
const params = new URLSearchParams({
client_id: this.config.clientId,
client_secret: this.config.clientSecret,
code,
grant_type: 'authorization_code',
redirect_uri: this.config.redirectUri,
});

      const response = await fetch(this.GOOGLE_TOKEN_URL, {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: params.toString(),
        timeout: 10000, // 10 second timeout
      });

      if (!response.ok) {
        this.logger.error(
          `Google token exchange failed: ${response.status} ${response.statusText}`
        );
        throw new Error(
          `Google token exchange failed: ${response.statusText}`
        );
      }

      const data = (await response.json()) as GoogleTokenResponse;

      if (!data.access_token) {
        this.logger.error('Google token response missing access_token');
        throw new Error('Token response missing access_token');
      }

      this.logger.debug('Successfully exchanged code for access token');

      return data;
    } catch (error) {
      this.logger.error(`Failed to exchange authorization code: ${error}`);
      throw new Error('Failed to exchange authorization code');
    }

}

/**

- Retrieve user profile from Google
-
- Makes HTTP request to Google userinfo endpoint.
- Retrieves email, name, picture, etc.
-
- @param accessToken - Google access token
- @returns Google user profile
- @throws Error if profile retrieval fails
-
- @example
- ```typescript

  ```
- const profile = await oauthService.getUserProfile(accessToken);
- console.log(profile.email); // User's Google email
- ```

  ```

*/
private async getUserProfile(
accessToken: string
): Promise<GoogleUserInfoResponse> {
try {
const response = await fetch(this.GOOGLE_USERINFO_URL, {
method: 'GET',
headers: {
Authorization: `Bearer ${accessToken}`,
'Accept': 'application/json',
},
timeout: 10000, // 10 second timeout
});

      if (!response.ok) {
        this.logger.error(
          `Google userinfo failed: ${response.status} ${response.statusText}`
        );
        throw new Error(
          `Google userinfo failed: ${response.statusText}`
        );
      }

      const profile = (await response.json()) as GoogleUserInfoResponse;

      if (!profile.email) {
        this.logger.error('Google profile missing email');
        throw new Error('Profile missing required email field');
      }

      this.logger.debug(`Retrieved profile for Google user: ${profile.sub}`);

      return profile;
    } catch (error) {
      this.logger.error(`Failed to retrieve user profile: ${error}`);
      throw new Error('Failed to retrieve user profile');
    }

}

/**

- Convert Google profile to VeriFit user format
-
- Maps Google userinfo → AuthenticatedUser
-
- @param googleProfile - Google user profile
- @param role - User role (from database pre-creation)
- @param organizationId - Organization ID (from database)
- @returns VeriFit AuthenticatedUser
  */
  private mapGoogleProfileToUser(
  googleProfile: GoogleUserInfoResponse,
  role: 'ADMIN' | 'STAFF' | 'STUDENT',
  organizationId: string,
  userId: string
  ): AuthenticatedUser {
  return {
  id: userId, // Database user ID
  email: googleProfile.email,
  organizationId,
  role,
  lastLoginAt: new Date(),
  accountStatus: 'ACTIVE',
  };
  }

/**

- Handle OAuth callback
-
- Complete OAuth flow:
- 1.  Verify state (CSRF protection)
- 2.  Exchange code for access token
- 3.  Retrieve user profile
- 4.  Lookup user in database (pre-created)
- 5.  Generate JWT
- 6.  Create login response
-
- @param request - Callback request (code + state from Google)
- @param getUserFromDatabase - Function to lookup user by email
- @returns LoginResponse (JWT + user info)
- @throws BadRequestException if any step fails
-
- @example
- ```typescript

  ```
- const loginResponse = await oauthService.handleCallback(
- { code: '...', state: '...' },
- async (email) => {
-     const user = await db.user.findUnique({ where: { email } });
-     return user;
- }
- );
- // Response contains JWT + user info
- ```

  ```

*/
async handleCallback(
request: OAuthCallbackRequest,
getUserFromDatabase: (email: string) => Promise<{
id: string;
email: string;
role: 'ADMIN' | 'STAFF' | 'STUDENT';
organizationId: string;
accountStatus: 'ACTIVE' | 'LOCKED' | 'SUSPENDED';
} | null>
): Promise<LoginResponse> {
// Validate input
try {
OAuthCallbackRequestSchema.parse(request);
} catch (error) {
this.logger.warn(`OAuth callback validation failed: ${error}`);
throw new BadRequestException('Invalid callback parameters');
}

    // Step 1: Verify state (CSRF protection)
    const oauthState = this.verifyOAuthState(request.state);

    try {
      // Step 2: Exchange code for access token
      const tokens = await this.exchangeCodeForToken(request.code);

      // Step 3: Retrieve user profile from Google
      const googleProfile = await this.getUserProfile(tokens.access_token);

      // Step 4: Lookup user in database (pre-created by admin)
      const user = await getUserFromDatabase(googleProfile.email);

      if (!user) {
        this.logger.warn(
          `User not found for email: ${googleProfile.email}. User must be pre-created.`
        );
        throw new BadRequestException(
          'User not found. Please contact administrator.'
        );
      }

      // Check account status
      if (user.accountStatus === 'LOCKED') {
        this.logger.warn(`Login attempted on locked account: ${user.email}`);
        throw new BadRequestException(
          'Account is locked. Please contact administrator.'
        );
      }

      if (user.accountStatus === 'SUSPENDED') {
        this.logger.warn(
          `Login attempted on suspended account: ${user.email}`
        );
        throw new BadRequestException('Account suspended.');
      }

      // Step 5: Generate JWT
      const jwtToken = this.jwtService.generateToken({
        userId: user.id,
        email: user.email,
        organizationId: user.organizationId,
        role: user.role,
      });

      // Step 6: Map to authenticated user
      const authenticatedUser = this.mapGoogleProfileToUser(
        googleProfile,
        user.role,
        user.organizationId,
        user.id
      );

      // Get token expiration
      const expirationDate = this.jwtService.getExpirationDate(jwtToken);
      if (!expirationDate) {
        throw new Error('Failed to determine token expiration');
      }

      const loginResponse: LoginResponse = {
        token: jwtToken,
        user: authenticatedUser,
        expiresAt: Math.floor(expirationDate.getTime() / 1000), // unix timestamp
        redirectUrl: '/dashboard',
      };

      this.logger.log(`Successful login for user: ${user.email}`);

      return loginResponse;
    } catch (error) {
      // Don't re-throw BadRequestException (already thrown above)
      if (error instanceof BadRequestException) {
        throw error;
      }

      // Log other errors without exposing details to client
      this.logger.error(`OAuth callback failed: ${error}`);

      throw new BadRequestException(
        'Login failed. Please try again or contact support.'
      );
    }

}

/**

-    Clean up expired states periodically
-
-    Prevents memory leaks from accumulating old states
-    Called after every generateAuthorizationUrl
-
-    @private
     */
     private cleanupExpiredStates(): void {
     const now = new Date();
     let deleted = 0;

    for (const [state, oauthState] of this.oauthStates.entries()) {

      if (oauthState.expiresAt < now) {
        this.oauthStates.delete(state);
        deleted++;
      }
    }

    if (deleted > 0) {
      this.logger.debug(`Cleaned up ${deleted} expired OAuth states`);
    }

}

/**

- Get current number of active OAuth states
- Use for monitoring/debugging only
-
- @returns Number of active states in memory
  */
  getActiveStatesCount(): number {
  this.cleanupExpiredStates();
  return this.oauthStates.size;
  }

Task 1.3 review

1. node-fetch is wrong for our baseline ❌
   The code imports:

import fetch from 'node-fetch';
but Step 5 itself says Node 18+ already provides fetch.

Our Node baseline is Node 24, so adding node-fetch is unnecessary dependency/API duplication.

Required change: use the native global fetch and do not add node-fetch.

2. The timeout: 10000 option is not valid native fetch behavior ❌
   The code does:

fetch(url, {
...
timeout: 10000,
});
Native Node fetch uses AbortSignal, not a timeout option in the Fetch API.

So the implementation needs:

const controller = new AbortController();
const timeout = setTimeout(() => controller.abort(), 10_000);

try {
await fetch(url, { signal: controller.signal });
} finally {
clearTimeout(timeout);
}
Or an equivalent reusable timeout helper.

This is important because the document explicitly requires bounded network calls.

3. It claims Google issuer verification but does not actually do it ❌
   The header says:

Verify issuer (iss) matches Google

But the implementation never verifies an OIDC ID token's:

iss

aud

exp

nonce

It simply exchanges the authorization code and calls Google's userinfo endpoint.

So the security documentation and implementation disagree.

Required change: decide on one explicit identity-validation model.

For a production-quality Google OIDC integration, I would require the Google identity token to be validated against the expected Google issuer/audience, including the OAuth client ID, and bind the authenticated identity to the verified Google sub.

4. User lookup by email alone is too weak ❌
   Current flow:

getUserFromDatabase(googleProfile.email)
The stable Google identity is sub, not the display email.

For VeriFit, the safer long-term contract is:

Google issuer + Google subject (sub)
↓
linked VeriFit identity
↓
local User
Email can be a useful attribute, but it should not be the sole external identity binding.

Required change: Task 1.3 should define an identity lookup contract that supports provider + provider subject, rather than making email the permanent identity key.

This may require a small database/schema decision, which should be made before implementation rather than hidden inside the OAuth service.

5. AuthenticatedUser reintroduces fields we explicitly rejected earlier ❌
   The service creates:

lastLoginAt: new Date(),
accountStatus: 'ACTIVE',
and expects the database lookup to return:

accountStatus: 'ACTIVE' | 'LOCKED' | 'SUSPENDED';
But our current database baseline does not establish those fields.

We explicitly rejected freezing nonexistent accountStatus / lastLoginAt fields during 1.1/1.2.

So this task is trying to smuggle a new user-account model into OAuth.

Required change: OAuth must consume the actual current User model and not invent account-state fields.

If account status is needed, that must be a separate approved database task.

6. OAuth service should not silently invent the dashboard redirect ❌
   This is hardcoded:

redirectUrl: '/dashboard',
There is no product-level decision in the locked baseline that every successful login goes to /dashboard.

The frontend may eventually need an explicit post-login destination contract.

Required change: don't hardcode product navigation into the authentication domain. Either remove it from LoginResponse for now or make the destination an explicitly validated application-level input/configuration.

7. LoginInitiateRequestSchema is unused ❌
   The document defines:

const LoginInitiateRequestSchema = ...
but the actual public method is:

generateAuthorizationUrl(organizationId?: string)
The schema never participates in the flow.

That is dead API surface.

Also this field is particularly concerning:

redirectUri: z.string().url().optional()
because the service itself uses:

this.config.redirectUri
and never uses the request's redirect URI.

Required change: remove the unused request schema, or actually define a coherent request contract. Do not leave contradictory redirect-URI mechanisms.

8. organizationId is stored but not actually bound to the authentication transaction ⚠️
   This:

generateAuthorizationUrl(organizationId?: string)
puts the org ID into in-memory state, but the callback ultimately looks up the user only by email.

So the organization context does not meaningfully protect the login flow.

That can become dangerous in a multi-tenant system.

Required change: either defer organization-specific OAuth binding until multi-tenancy is designed, or explicitly carry and validate organization context through the identity lookup.

Don't expose a parameter that gives the appearance of tenant isolation without actually enforcing it.

9. In-memory OAuth state is acceptable only as a very explicit development adapter ⚠️
   The code says:

private oauthStates: Map<string, OAuthState> = new Map();
and:

Phase 6: migrate to Redis

For our architecture, this is not suitable as the actual production authentication-state mechanism.

Problems:

state disappears on process restart

doesn't work correctly with multiple API instances

no durable TTL

state cannot be shared across instances

process memory becomes the storage boundary

The bigger issue is that this is described as part of the service rather than an injected storage boundary.

Better design for 1.3:

OAuthService
↓
OAuthStateStore interface
↓
InMemoryOAuthStateStore (tests/dev)
Then a later infrastructure implementation can use Valkey/Redis without rewriting OAuth business logic.

That is much cleaner with our modular-monolith architecture.

10. State consumption needs an atomic single-use contract ⚠️
    The current implementation does:

const oauthState = this.oauthStates.get(state);
...
this.oauthStates.delete(state);
inside one process, which is effectively okay for the simple map, but when we move to Redis the semantics must explicitly become:

consume(state)
rather than:

get(state)
delete(state)
Otherwise concurrent callbacks can race.

Task 1.3 should define the single-use consumption contract now, even if its first implementation is in memory.

11. Logging leaks more authentication data than necessary ⚠️
    Examples:

Generated authorization URL for state: ${state}
and:

Successful login for user: ${user.email}
The state value is security-sensitive transaction material and should not normally be logged verbatim.

OAuth authorization codes and access tokens are not logged, which is good, but the logging policy should explicitly avoid:

authorization codes

access tokens

ID tokens

state values

raw bearer tokens

Email/user ID logging should be carefully controlled too.

12. access_type=offline + prompt=consent is unnecessary for the current product scope ⚠️
    The task does not implement Google API access or refresh-token management.

Yet it requests:

access_type=offline
prompt=consent
That asks for more authorization behavior than we currently need.

For a simple authentication-only OIDC flow, request the minimum required scopes and behavior.

Required change: don't request offline access unless we have an approved refresh-token use case.

13. Google response bodies should be schema-validated ❌
    The code does:

const data = (await response.json()) as GoogleTokenResponse;
and:

const profile = (await response.json()) as GoogleUserInfoResponse;
That's only a TypeScript assertion. Runtime data is completely unvalidated.

We already have Zod in the workspace.

Required change: use Zod schemas for Google's token response and userinfo response.

This is especially important because Task 1.3 is an external-input boundary.

14. The tests are not production-quality yet ❌
    There are multiple as any casts in the supplied tests.

We already explicitly removed any from the 1.2 test strategy.

Also this test:

it('should reject expired state', ...)
doesn't actually test an expired state. The code itself admits it:

"we'll test by manually waiting"

followed by testing an unknown state.

That's not testing expiration.

And this:

const testConfig = ...
does nothing useful.

Required tests instead:

actual expired-state behavior using fake timers or injected clock

state is atomically single-use

state cannot be reused

timeout causes fetch cancellation

malformed Google token response rejected

malformed userinfo response rejected

wrong/missing issuer rejected

wrong audience rejected

unverified identity data cannot create a local session

Google sub binding

all network failures handled without leaking provider secrets

no access token/state/code appears in logs

native fetch, no node-fetch

15. The callback has too many responsibilities ⚠️
    handleCallback() currently handles:

validation
→ state
→ Google token exchange
→ Google profile retrieval
→ local user lookup
→ account status
→ JWT generation
→ authenticated-user mapping
→ response construction
That's workable for a small service, but it's becoming a god method.

We should keep it understandable by separating the provider/network operations from local authentication decisions.

16. OAuthService should not create the appearance of database ownership ❌
    This part:

getUserFromDatabase: (email) => ...
is actually a good instinct because it avoids coupling the auth package directly to Prisma.

But the contract itself needs to be upgraded from:

getUserFromDatabase(email)
to an explicit identity lookup boundary.

For example conceptually:

OAuth identity
↓
UserIdentityResolver
↓
local User/Auth identity
Then @verifit/auth doesn't know Prisma models.

That preserves our existing package boundary.

One especially important contradiction
The document says the OAuth service should:

"Create LoginResponse with JWT + user info"

and LoginResponse contains:

token: string;
while the earlier authentication security direction was to avoid unnecessary raw token persistence and prefer secure session/token handling.

Returning the JWT is not inherently wrong, but the comment says:

/** JWT token (sent in httpOnly cookie, also in response) */
That is a bad contract.

Do not send the same bearer credential both in an HTTP-only cookie and response JSON.

Pick the authentication transport deliberately in the API layer. The auth domain service should not bake both mechanisms into one response type.
