/**
 * Auth Guard - JWT Validation
 *
 * Responsibilities:
 * - Extract JWT from HTTP-only cookie or Authorization header
 * - Validate JWT signature and expiration
 * - Extract user claims from JWT
 * - Attach user to request (req.user)
 * - Reject requests without valid JWT (401 Unauthorized)
 *
 * Usage:
 * @UseGuards(AuthGuard)
 * async getStudents() { ... }
 *
 * Security:
 * - JWT must be valid and not expired
 * - Token is verified against JWT_SECRET
 * - User info is extracted from verified JWT payload
 * - Revocation check is deferred to AuthorizationService (Phase 2)
 */

import {
  Injectable,
  CanActivate,
  ExecutionContext,
  UnauthorizedException,
  Logger,
} from '@nestjs/common';
import { Request } from 'express';
import { JwtService } from '../services/jwt.service';
import { JwtPayload, AuthenticatedUser } from '../types';

/**
 * Auth Guard - validates JWT on protected routes
 *
 * @example
 * ```typescript
 * @Controller('api/v1/students')
 * export class StudentController {
 *   @Get()
 *   @UseGuards(AuthGuard)
 *   async list(@Request() req) {
 *     // req.user is set by AuthGuard
 *     console.log('Accessing as:', req.user.email);
 *   }
 * }
 * ```
 */
@Injectable()
export class AuthGuard implements CanActivate {
  private readonly logger = new Logger(AuthGuard.name);

  constructor(private readonly jwtService: JwtService) {}

  /**
   * Validate JWT and attach user to request
   *
   * Flow:
   * 1. Extract JWT from cookie or Authorization header
   * 2. Validate JWT (signature + expiration)
   * 3. Extract user claims from payload
   * 4. Attach to req.user
   * 5. Return true (allow request to proceed)
   *
   * If any step fails, throw UnauthorizedException (401)
   *
   * @param context - NestJS execution context
   * @returns true if authorized, throws otherwise
   */
  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest<Request>();

    // Step 1: Extract JWT
    const token = this.extractTokenFromRequest(request);

    if (!token) {
      this.logger.warn(
        `Request without JWT: ${request.method} ${request.path}`
      );
      throw new UnauthorizedException('JWT token required');
    }

    // Step 2: Validate JWT
    const payload = this.jwtService.verifyToken(token);

    if (!payload) {
      this.logger.warn(`Invalid JWT on ${request.method} ${request.path}`);
      throw new UnauthorizedException('Invalid or expired token');
    }

    // Step 3: Verify payload structure (should always have these)
    if (!payload.userId || !payload.organizationId || !payload.role) {
      this.logger.error(
        `JWT payload missing required claims: ${JSON.stringify(payload)}`
      );
      throw new UnauthorizedException('Invalid token claims');
    }

    // Step 4: Map JWT payload to AuthenticatedUser
    const user: AuthenticatedUser = {
      id: payload.userId,
      email: payload.email,
      organizationId: payload.organizationId,
      role: payload.role,
      accountStatus: 'ACTIVE', // TODO: check database for actual status (Phase 1.5)
    };

    // Step 5: Attach to request
    request.user = user as any;

    this.logger.debug(
      `JWT validated for user ${user.id} on ${request.method} ${request.path}`
    );

    return true;
  }

  /**
   * Extract JWT from request
   *
   * Tries in order:
   * 1. HTTP-only cookie (preferred for web browsers)
   * 2. Authorization header with "Bearer " prefix (for API clients)
   *
   * @param request - Express request
   * @returns JWT token string, or undefined if not found
   *
   * @private
   */
  private extractTokenFromRequest(request: Request): string | undefined {
    // Try cookie first (HTTP-only, CSRF protected)
    if (request.cookies?.jwt) {
      return request.cookies.jwt;
    }

    // Try Authorization header (for API clients / mobile)
    const authHeader = request.headers.authorization;
    if (authHeader?.startsWith('Bearer ')) {
      return authHeader.substring(7);
    }

    return undefined;
  }
}

/**
 * Role Guard - Role-Based Access Control
 *
 * Responsibilities:
 * - Check if user has required role
 * - Support multiple allowed roles
 * - Reject if user's role insufficient (403 Forbidden)
 * - Must be used with AuthGuard (assumes req.user exists)
 *
 * Usage:
 * @UseGuards(AuthGuard, RoleGuard)
 * @AllowRoles('ADMIN', 'STAFF')
 * async updateStudent() { ... }
 *
 * Security:
 * - Default deny (no access unless role explicitly allowed)
 * - Requires AuthGuard to run first
 * - Role comes from verified JWT payload
 */

import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Logger,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { Request } from 'express';
import { AuthenticatedUser, UserRole } from '../types';

/**
 * Metadata key for allowed roles
 * Set via @AllowRoles decorator
 */
export const ALLOWED_ROLES_KEY = 'allowed_roles';

/**
 * Role Guard - checks if user has allowed role
 *
 * @example
 * ```typescript
 * @Controller('api/v1/students')
 * export class StudentController {
 *   @Post()
 *   @UseGuards(AuthGuard, RoleGuard)
 *   @AllowRoles('ADMIN', 'STAFF')
 *   async create(@Body() dto: CreateStudentDto) {
 *     // Only ADMIN or STAFF can reach here
 *   }
 *
 *   @Get()
 *   @UseGuards(AuthGuard) // RoleGuard not needed (public to all auth'd users)
 *   async list() {
 *     // All authenticated users can list
 *   }
 * }
 * ```
 */
@Injectable()
export class RoleGuard implements CanActivate {
  private readonly logger = new Logger(RoleGuard.name);

  constructor(private reflector: Reflector) {}

  /**
   * Check if user's role is in allowed roles
   *
   * Flow:
   * 1. Get allowed roles from @AllowRoles decorator
   * 2. If no decorator, allow all authenticated users
   * 3. Check user's role against allowed roles
   * 4. Return true if match, throw 403 if not
   *
   * @param context - NestJS execution context
   * @returns true if authorized, throws otherwise
   */
  canActivate(context: ExecutionContext): boolean {
    // Step 1: Get allowed roles from decorator
    const allowedRoles = this.reflector.get<UserRole[]>(
      ALLOWED_ROLES_KEY,
      context.getHandler()
    );

    // Step 2: If no decorator, allow all authenticated users
    if (!allowedRoles || allowedRoles.length === 0) {
      return true; // No role restriction, already authenticated by AuthGuard
    }

    // Step 3: Get user from request (set by AuthGuard)
    const request = context.switchToHttp().getRequest<Request>();
    const user = request.user as AuthenticatedUser | undefined;

    if (!user) {
      this.logger.error(
        'RoleGuard called without user in request. AuthGuard must run first.'
      );
      throw new ForbiddenException('User not found in request context');
    }

    // Step 4: Check if user's role is allowed
    if (!allowedRoles.includes(user.role)) {
      this.logger.warn(
        `Role check failed: user ${user.id} has ${user.role}, required ${allowedRoles.join(' or ')}`
      );
      throw new ForbiddenException(
        `Insufficient role. Required: ${allowedRoles.join(' or ')}, have: ${user.role}`
      );
    }

    this.logger.debug(
      `Role check passed for user ${user.id} (${user.role}) on ${request.method} ${request.path}`
    );

    return true;
  }
}

/**
 * Org Guard - Organization Scoping (IDOR Prevention)
 *
 * Responsibilities:
 * - Check if user can access resource in their organization
 * - Prevent cross-org access (IDOR attacks)
 * - Extract resource org from route params or body
 * - Allow ADMIN cross-org access
 *
 * Usage:
 * @UseGuards(AuthGuard, OrgGuard)
 * async getStudent(@Param('organizationId') orgId: string) { ... }
 *
 * Security:
 * - Validates organizationId from request against user's org
 * - ADMIN can bypass org scoping
 * - STAFF/STUDENT cannot access other orgs
 * - Returns 403 for cross-org attempts (IDOR prevention)
 */

import {
  Injectable,
  CanActivate,
  ExecutionContext,
  ForbiddenException,
  Logger,
  BadRequestException,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { Request } from 'express';
import { AuthenticatedUser } from '../types';

/**
 * Metadata key for org ID source
 * Set via @OrgIdSource decorator
 *
 * Possible values:
 * - 'param:organizationId' - extract from route param
 * - 'param:orgId' - alternative route param
 * - 'body:organizationId' - extract from request body
 * - null/undefined - use user's org (default)
 */
export const ORG_ID_SOURCE_KEY = 'org_id_source';

/**
 * Org Guard - prevents cross-org access (IDOR prevention)
 *
 * @example
 * ```typescript
 * @Controller('api/v1')
 * export class StudentController {
 *   @Get('organizations/:organizationId/students/:studentId')
 *   @UseGuards(AuthGuard, OrgGuard)
 *   @OrgIdSource('param:organizationId')
 *   async getStudent(
 *     @Param('organizationId') orgId: string,
 *     @Param('studentId') studentId: string
 *   ) {
 *     // Guard ensures orgId matches user's org (unless ADMIN)
 *   }
 *
 *   @Post('students')
 *   @UseGuards(AuthGuard, OrgGuard)
 *   async create(@Body() dto: CreateStudentDto) {
 *     // No @OrgIdSource = uses user's org from JWT
 *   }
 * }
 * ```
 */
@Injectable()
export class OrgGuard implements CanActivate {
  private readonly logger = new Logger(OrgGuard.name);

  constructor(private reflector: Reflector) {}

  /**
   * Check if user can access resource in organization
   *
   * Flow:
   * 1. Get user from request (set by AuthGuard)
   * 2. If ADMIN, allow cross-org access
   * 3. Get resource organization ID (from param, body, or user's org)
   * 4. Compare with user's org
   * 5. Deny if mismatch (IDOR prevention)
   *
   * @param context - NestJS execution context
   * @returns true if authorized, throws otherwise
   */
  canActivate(context: ExecutionContext): boolean {
    // Step 1: Get user
    const request = context.switchToHttp().getRequest<Request>();
    const user = request.user as AuthenticatedUser | undefined;

    if (!user) {
      this.logger.error(
        'OrgGuard called without user in request. AuthGuard must run first.'
      );
      throw new ForbiddenException('User not found in request context');
    }

    // Step 2: ADMIN bypass (can access any org)
    if (user.role === 'ADMIN') {
      this.logger.debug(
        `Org bypass for ADMIN ${user.id} on ${request.method} ${request.path}`
      );
      return true;
    }

    // Step 3: Get resource org ID
    const resourceOrgId = this.extractResourceOrgId(context, request, user);

    if (!resourceOrgId) {
      this.logger.error(
        `Failed to extract resource org ID from ${request.method} ${request.path}`
      );
      throw new BadRequestException('Organization ID required');
    }

    // Step 4: Compare with user's org
    if (user.organizationId !== resourceOrgId) {
      this.logger.warn(
        `Org scoping violation: user ${user.id} (org ${user.organizationId}) tried to access org ${resourceOrgId}`
      );
      throw new ForbiddenException('Cannot access resources in other organizations');
    }

    this.logger.debug(
      `Org scoping check passed for user ${user.id} on org ${resourceOrgId}`
    );

    return true;
  }

  /**
   * Extract resource organization ID from request
   *
   * Tries in order:
   * 1. Route parameter (if @OrgIdSource decorator set)
   * 2. Request body (if POST/PUT and has organizationId)
   * 3. User's org (fallback)
   *
   * @param context - NestJS execution context
   * @param request - Express request
   * @param user - Authenticated user
   * @returns Organization ID, or undefined if not found
   *
   * @private
   */
  private extractResourceOrgId(
    context: ExecutionContext,
    request: Request,
    user: AuthenticatedUser
  ): string | undefined {
    // Try decorator metadata
    const orgIdSource = this.reflector.get<string>(
      ORG_ID_SOURCE_KEY,
      context.getHandler()
    );

    if (orgIdSource) {
      const [source, key] = orgIdSource.split(':');

      if (source === 'param') {
        const paramValue = (request.params as Record<string, any>)?.[key];
        if (paramValue) {
          return paramValue;
        }
      }

      if (source === 'body') {
        const bodyValue = (request.body as Record<string, any>)?.[key];
        if (bodyValue) {
          return bodyValue;
        }
      }
    }

    // Try request body organizationId
    if (
      request.method === 'POST' ||
      request.method === 'PUT' ||
      request.method === 'PATCH'
    ) {
      const bodyOrgId = (request.body as Record<string, any>)?.organizationId;
      if (bodyOrgId) {
        return bodyOrgId;
      }
    }

    // Fallback: use user's org
    return user.organizationId;
  }
}

/**
 * Custom Authentication Decorators
 *
 * Decorators for common auth patterns:
 * - @AllowRoles() - specify allowed roles
 * - @OrgIdSource() - specify where org ID comes from
 * - @CurrentUser() - inject authenticated user
 */

import { SetMetadata } from '@nestjs/common';
import { UserRole } from '../types';
import {
  ALLOWED_ROLES_KEY,
  ORG_ID_SOURCE_KEY,
} from '../guards';

/**
 * Specify allowed roles for route
 *
 * @example
 * ```typescript
 * @Post('students')
 * @UseGuards(AuthGuard, RoleGuard)
 * @AllowRoles('ADMIN', 'STAFF')
 * async create() { ... }
 * ```
 */
export const AllowRoles = (...roles: UserRole[]) =>
  SetMetadata(ALLOWED_ROLES_KEY, roles);

/**
 * Specify where organization ID comes from in request
 *
 * @example
 * ```typescript
 * @Get('organizations/:organizationId/students/:studentId')
 * @UseGuards(AuthGuard, OrgGuard)
 * @OrgIdSource('param:organizationId')
 * async getStudent() { ... }
 *
 * // Or from body
 * @Post('students')
 * @UseGuards(AuthGuard, OrgGuard)
 * @OrgIdSource('body:organizationId')
 * async create() { ... }
 * ```
 */
export const OrgIdSource = (source: string) =>
  SetMetadata(ORG_ID_SOURCE_KEY, source);

/**
 * Note: @CurrentUser() decorator would inject user from request

/**
 * Auth Guard Tests
 *
 * Coverage:
 * - Valid JWT (allowed)
 * - Expired JWT (rejected)
 * - Invalid JWT (rejected)
 * - Missing JWT (rejected)
 * - JWT from cookie (allowed)
 * - JWT from Authorization header (allowed)
 * - Token extraction and validation
 */

import { describe, it, expect, beforeEach, vi } from 'vitest';
import { UnauthorizedException, ExecutionContext } from '@nestjs/common';
import { AuthGuard } from './auth.guard';
import { JwtService, JwtConfig } from '../services/jwt.service';

describe('AuthGuard', () => {
  let guard: AuthGuard;
  let jwtService: JwtService;

  const jwtConfig: JwtConfig = {
    secret: 'test-secret-key-minimum-32-characters-long-for-testing',
    expiresIn: '15m',
  };

  const testUser = {
    userId: 'user-123',
    email: 'test@example.com',
    organizationId: 'org-456',
    role: 'STUDENT' as const,
  };

  beforeEach(() => {
    jwtService = new JwtService(jwtConfig);
    guard = new AuthGuard(jwtService);
  });

  describe('canActivate', () => {
    it('should allow valid JWT from cookie', async () => {
      const token = jwtService.generateToken(testUser);

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: { jwt: token },
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      const result = await guard.canActivate(mockContext);

      expect(result).toBe(true);
    });

    it('should allow valid JWT from Authorization header', async () => {
      const token = jwtService.generateToken(testUser);

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: {},
            headers: { authorization: `Bearer ${token}` },
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      const result = await guard.canActivate(mockContext);

      expect(result).toBe(true);
    });

    it('should prefer cookie over Authorization header', async () => {
      const token1 = jwtService.generateToken(testUser);
      const token2 = jwtService.generateToken({
        ...testUser,
        userId: 'other-user',
      });

      const mockRequest = {
        cookies: { jwt: token1 },
        headers: { authorization: `Bearer ${token2}` },
        method: 'GET',
        path: '/api/v1/students',
        user: undefined,
      };

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => mockRequest,
        }),
      } as ExecutionContext;

      const result = await guard.canActivate(mockContext);

      expect(result).toBe(true);
      expect(mockRequest.user.id).toBe('user-123'); // From token1 (cookie)
    });

    it('should reject missing JWT', async () => {
      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: {},
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });

    it('should reject invalid JWT', async () => {
      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: { jwt: 'invalid.token.here' },
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });

    it('should reject expired JWT', async () => {
      const shortLivedService = new JwtService({
        secret: jwtConfig.secret,
        expiresIn: '1ms',
      });

      const token = shortLivedService.generateToken(testUser);

      // Wait for token to expire
      await new Promise((resolve) => setTimeout(resolve, 100));

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: { jwt: token },
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });

    it('should attach user to request', async () => {
      const token = jwtService.generateToken(testUser);

      const mockRequest = {
        cookies: { jwt: token },
        headers: {},
        method: 'GET',
        path: '/api/v1/students',
        user: undefined,
      };

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => mockRequest,
        }),
      } as ExecutionContext;

      await guard.canActivate(mockContext);

      expect(mockRequest.user).toBeDefined();
      expect(mockRequest.user.id).toBe(testUser.userId);
      expect(mockRequest.user.email).toBe(testUser.email);
      expect(mockRequest.user.organizationId).toBe(testUser.organizationId);
      expect(mockRequest.user.role).toBe(testUser.role);
      expect(mockRequest.user.accountStatus).toBe('ACTIVE');
    });

    it('should reject JWT with missing userId', async () => {

        organizationId: 'org-456',
        role: 'STUDENT',
        iat: Math.floor(Date.now() / 1000),
        exp: Math.floor(Date.now() / 1000) + 900,
      };

      const invalidToken = require('jsonwebtoken').sign(
        invalidPayload,
        jwtConfig.secret,
        { algorithm: 'HS256' }
      );

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: { jwt: invalidToken },
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });

    it('should reject JWT with missing organizationId', async () => {
      const invalidPayload = {
        userId: 'user-123',
        email: 'test@example.com',
        role: 'STUDENT',
        iat: Math.floor(Date.now() / 1000),
        exp: Math.floor(Date.now() / 1000) + 900,
      };

      const invalidToken = require('jsonwebtoken').sign(
        invalidPayload,
        jwtConfig.secret,
        { algorithm: 'HS256' }
      );

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: { jwt: invalidToken },
            headers: {},
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });
  });

  describe('token extraction', () => {
    it('should extract JWT from cookie with key "jwt"', async () => {
      const token = jwtService.generateToken(testUser);

      const mockRequest = {
        cookies: { jwt: token },
        headers: {},
        method: 'GET',
        path: '/api/v1/students',
        user: undefined,
      };

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => mockRequest,
        }),
      } as ExecutionContext;

      await guard.canActivate(mockContext);

      expect(mockRequest.user).toBeDefined();
    });

    it('should extract JWT from Authorization header with Bearer scheme', async () => {
      const token = jwtService.generateToken(testUser);

      const mockRequest = {
        cookies: {},
        headers: { authorization: `Bearer ${token}` },
        method: 'GET',
        path: '/api/v1/students',
        user: undefined,
      };

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => mockRequest,
        }),
      } as ExecutionContext;

      await guard.canActivate(mockContext);

      expect(mockRequest.user).toBeDefined();
    });

    it('should reject Authorization header without Bearer prefix', async () => {
      const token = jwtService.generateToken(testUser);

      const mockContext = {
        switchToHttp: () => ({
          getRequest: () => ({
            cookies: {},
            headers: { authorization: `Basic ${token}` },
            method: 'GET',
            path: '/api/v1/students',
          }),
        }),
      } as ExecutionContext;

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );

    it('should reject malformed Authorization header', async () => {

      await expect(guard.canActivate(mockContext)).rejects.toThrow(
        UnauthorizedException
      );
    });

  describe('error handling', () => {
    it('should throw UnauthorizedException for expired token', async () => {

hecked the actual Task 1.5 document end-to-end. The direction is reasonable, but I would not give this version to the code giver yet.

There are several important issues, and two are especially serious because they undo decisions we just locked in 1.4.

🚨 MUST FIX BEFORE IMPLEMENTATION
1. OrgGuard trusts client-supplied organization IDs ❌
This is the biggest problem.

The document explicitly says the OrgGuard should extract the organization ID from:

route params
request body
user's org
and even defines:

@OrgIdSource('body:organizationId')


Then the implementation reads:

(request.body as Record<string, any>)?.[key]
and treats that as the resource organization. 


This directly conflicts with the 1.4 invariant we just approved:

Resource ownership / organization must come from trusted server-side resource data, not from attacker-controlled request data.

A malicious request can simply send:

organizationId = attacker's-org
while referencing a resource belonging to another organization.

Required change
Do not make OrgGuard responsible for discovering authoritative resource ownership from request body/params.

It should only validate a trusted organization context that the application has established.

For example:

request
  ↓
controller/service resolves resource
  ↓
trusted ResourceContext
  ├── resourceId
  ├── organizationId
  └── ownerUserId (when applicable)
  ↓
AuthorizationService
Task 1.5 should not create a shortcut around that.

2. The OrgGuard doesn't actually prevent IDOR on resource IDs ❌
The task claims:

"Prevent IDOR attacks at the route level."

But an organization check alone doesn't prove that a requested resource belongs to that organization.

Example:

GET /students/student-999
If the guard only knows:

user.organizationId = org-1
request organizationId = org-1
it has no idea whether student-999 actually belongs to org-1.

This is precisely why we changed 1.4 to require trusted resource ownership.

Required change
Don't claim that OrgGuard itself provides complete IDOR protection for resource IDs.

The responsibilities should be:

AuthGuard
→ Is the caller authenticated?

RoleGuard
→ Is this role permitted?

AuthorizationService
→ Is this specific user allowed to access this specific trusted resource?

OrgGuard
→ Optional route-level tenant-context consistency check.
Actual resource ownership remains a server-side data concern.

3. AuthGuard reintroduces accountStatus ❌
The code contains:

accountStatus: 'ACTIVE',


with:

TODO: check database for actual status

We explicitly removed this assumption from OAuth/1.4 because that field is not established in the current database model.

Do not put it back.

The authenticated request principal should contain only information actually established by the JWT/authentication contract.

4. The claim "revocation deferred to AuthorizationService" is architecturally wrong ❌
The document says:

Revocation check is deferred to AuthorizationService (Phase 2). 


AuthorizationService should not be responsible for JWT/session revocation.

These are separate concerns:

Authentication
  ├── signature
  ├── expiry
  └── revocation/session validity

Authorization
  ├── role
  ├── tenant
  └── resource permission
Revocation belongs in the authentication/session layer, not authorization.

Task 1.6's database/session work may introduce the persistence mechanism, but don't design 1.5 around the wrong ownership.

5. Cookie authentication introduces a CSRF problem that the spec ignores ⚠️
The AuthGuard prefers:

HTTP-only cookie
and describes it as:

HTTP-only, CSRF protected. 


HTTP-only does not itself provide CSRF protection.

If authentication credentials are automatically attached via cookies, CSRF protections need to be deliberately designed at the API layer.

So don't document:

HTTP-only = CSRF protected
as a security invariant.

For 1.5, either:

A. make cookie authentication a later explicitly designed feature, or

B. define the required CSRF strategy as part of the API authentication transport.

Don't leave a false security assumption in the code comments.

6. Cookie + Authorization header precedence needs a deliberate policy ⚠️
The guard currently does:

cookie first
then Authorization header


That creates ambiguity when both credentials are supplied.

Example:

Cookie → User A
Authorization → User B
Which identity should the application trust?

The spec chooses cookie, but there is no product/security justification.

Better
Define one explicit authentication transport policy.

For example:

Browser session → cookie
API bearer authentication → Authorization
and reject conflicting credentials rather than silently choosing one.

At minimum, test the conflict case.

7. request.user as any violates our baseline ❌
The implementation does:

request.user = user as any;


We have repeatedly established no any.

9. RoleGuard default-allow behavior needs tightening ⚠️
