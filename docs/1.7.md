typescript
/**
 * Auth Module - NestJS Authentication & Authorization
 *
 * Exports:
 * - JwtService (token generation/validation)
 * - OAuthService (Google OAuth login)
 * - AuthorizationService (RBAC + org scoping)
 * - AuthGuard, RoleGuard, OrgGuard (route protection)
 * - Decorators (@AllowRoles, @OrgIdSource)
 *
 * Usage in other modules:
 * ```typescript
 * @Module({
 *   imports: [AuthModule],
 * })
 * export class StudentModule { }
 * ```
 *
 * Usage in controllers:
 * ```typescript
 * @Controller('students')
 * export class StudentController {
 *   constructor(private authz: AuthorizationService) {}
 *
 *   @Get()
 *   @UseGuards(AuthGuard, RoleGuard)
 *   @AllowRoles('STAFF', 'ADMIN')
 *   async list(@Request() req) {
 *     // User authenticated and has required role
 *   }
 * }
 * ```
 */

import { Module, Global, DynamicModule, ModuleMetadata } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { JwtService, JwtConfig } from './services/jwt.service';
import { OAuthService, GoogleOAuthConfig } from './services/oauth.service';
import { AuthorizationService } from './services/authorization.service';
import { AuthGuard } from './guards/auth.guard';
import { RoleGuard } from './guards/role.guard';
import { OrgGuard } from './guards/org.guard';

/**
 * Auth Module Configuration
 * Loaded from environment variables
 */
export interface AuthModuleConfig {
  jwt: JwtConfig;
  oauth: GoogleOAuthConfig;
}

/**
 * Auth Module Options
 * For custom configuration or testing
 */
export interface AuthModuleOptions {
  jwt?: Partial<JwtConfig>;
  oauth?: Partial<GoogleOAuthConfig>;
  isGlobal?: boolean;
}

/**
 * Auth Module - Provides authentication and authorization services
 *
 * Features:
 * - JWT token generation and validation
 * - Google OAuth login
 * - Role-based access control (RBAC)
 * - Organization scoping (multi-tenancy)
 * - IDOR prevention
 */
@Global() // Make services available application-wide
@Module({
  providers: [
    JwtService,
    OAuthService,
    AuthorizationService,
    AuthGuard,
    RoleGuard,
    OrgGuard,
  ],
  exports: [
    JwtService,
    OAuthService,
    AuthorizationService,
    AuthGuard,
    RoleGuard,
    OrgGuard,
  ],
})
export class AuthModule {
  /**
   * Register Auth Module with configuration
   *
   * @param options - Authentication options (jwt, oauth)
   * @returns Dynamic module
   *
   * @example
   * ```typescript
   * @Module({
   *   imports: [
   *     AuthModule.register({
   *       jwt: {
   *         secret: process.env.JWT_SECRET,
   *         expiresIn: process.env.JWT_EXPIRATION || '15m'
   *       },
   *       oauth: {
   *         clientId: process.env.OAUTH_CLIENT_ID,
   *         clientSecret: process.env.OAUTH_CLIENT_SECRET,
   *         redirectUri: process.env.OAUTH_REDIRECT_URI
   *       },
   *       isGlobal: true
   *     })
   *   ]
   * })
   * export class AppModule { }
   * ```
   */
  static register(options: AuthModuleOptions = {}): DynamicModule {
    return {
      module: AuthModule,
      global: options.isGlobal ?? true,
      providers: [
        {
          provide: JwtService,
          useFactory: (configService: ConfigService) => {
            const config: JwtConfig = {
              secret:
                options.jwt?.secret ||
                configService.getOrThrow('JWT_SECRET'),
              expiresIn:
                options.jwt?.expiresIn ||
                configService.get('JWT_EXPIRATION', '15m'),
            };
            return new JwtService(config);
          },
          inject: [ConfigService],
        },
        {
          provide: OAuthService,
          useFactory: (
            jwtService: JwtService,
            configService: ConfigService
          ) => {
            const config: GoogleOAuthConfig = {
              clientId:
                options.oauth?.clientId ||
                configService.getOrThrow('OAUTH_CLIENT_ID'),
              clientSecret:
                options.oauth?.clientSecret ||
                configService.getOrThrow('OAUTH_CLIENT_SECRET'),
              redirectUri:
                options.oauth?.redirectUri ||
                configService.getOrThrow('OAUTH_REDIRECT_URI'),
            };
            return new OAuthService(jwtService, config);
          },
          inject: [JwtService, ConfigService],
        },
        AuthorizationService,
        AuthGuard,
        RoleGuard,
        OrgGuard,
      ],
      exports: [
        JwtService,
        OAuthService,
        AuthorizationService,
        AuthGuard,
        RoleGuard,
        OrgGuard,
      ],
    };
  }

  /**
   * Register Auth Module asynchronously
   *
   * For when configuration comes from external sources (database, API, etc.)
   *
   * @param options - Factory function for options
   * @returns Dynamic module
   *
   * @example
   * ```typescript
   * @Module({
   *   imports: [
   *     AuthModule.registerAsync({
   *       useFactory: async (configService: ConfigService) => ({
   *         jwt: {
   *           secret: await getSecretFromVault(),
   *           expiresIn: '15m'
   *         },
   *         oauth: {
   *           clientId: configService.get('OAUTH_CLIENT_ID'),
   *           // ...
   *         }
   *       }),
   *       inject: [ConfigService]
   *     })
   *   ]
   * })
   * export class AppModule { }
   * ```
   */
  static registerAsync(
    options: {
      useFactory: (
        ...args: any[]
      ) => Promise<AuthModuleOptions> | AuthModuleOptions;
      inject?: any[];
    }
  ): DynamicModule {
    return {
      module: AuthModule,
      global: true,
      providers: [
        {
          provide: JwtService,
          useFactory: async (...args: any[]) => {
            const moduleOptions = await options.useFactory(...args);
            const config: JwtConfig = {
              secret: moduleOptions.jwt?.secret || '',
              expiresIn: moduleOptions.jwt?.expiresIn || '15m',
            };
            return new JwtService(config);
          },
          inject: options.inject || [],
        },
        {
          provide: OAuthService,
          useFactory: async (
            jwtService: JwtService,
            ...args: any[]
          ) => {
            const moduleOptions = await options.useFactory(...args);
            const config: GoogleOAuthConfig = {
              clientId: moduleOptions.oauth?.clientId || '',
              clientSecret: moduleOptions.oauth?.clientSecret || '',
              redirectUri: moduleOptions.oauth?.redirectUri || '',
            };
            return new OAuthService(jwtService, config);
          },
          inject: [JwtService, ...(options.inject || [])],
        },
        AuthorizationService,
        AuthGuard,
        RoleGuard,
        OrgGuard,
      ],
      exports: [
        JwtService,
        OAuthService,
        AuthorizationService,
        AuthGuard,
        RoleGuard,
        OrgGuard,
      ],
    };
  }
}

/**
 * Auth Controller - Authentication Endpoints
 *
 * Endpoints:
 * POST   /api/v1/auth/login         - Initiate OAuth login
 * GET    /api/v1/auth/callback      - Handle OAuth callback
 * POST   /api/v1/auth/logout        - Invalidate session
 * GET    /api/v1/auth/me            - Get current user
 * POST   /api/v1/auth/verify        - Verify token
 *
 * Note: This is an example controller that integrates auth services.
 * In production, move to apps/api/src/auth/auth.controller.ts
 * along with database integration (PrismaService).
 */

import {
  Controller,
  Get,
  Post,
  Query,
  Request,
  Response,
  UseGuards,
  BadRequestException,
  UnauthorizedException,
  Logger,
} from '@nestjs/common';
import { Response as ExpressResponse } from 'express';
import { JwtService } from './services/jwt.service';
import { OAuthService, OAuthCallbackRequest } from './services/oauth.service';
import { AuthGuard } from './guards/auth.guard';
import { z } from 'zod';

/**
 * Example Auth Controller
 *
 * IMPORTANT: This is a reference implementation.
 * Production implementation should:
 * 1. Live in apps/api/src/auth/auth.controller.ts
 * 2. Integrate with PrismaService (database)
 * 3. Use AuditService for logging
 * 4. Handle database user creation/updates
 */
@Controller('api/v1/auth')
export class AuthController {
  private readonly logger = new Logger(AuthController.name);

  constructor(
    private readonly jwtService: JwtService,
    private readonly oauthService: OAuthService
  ) {}

  /**
   * POST /api/v1/auth/login
   *
   * Initiate OAuth login (redirect to Google)
   *
   * @param res - Express response (for redirect)
   * @returns Redirect to Google OAuth consent screen
   */
  @Post('login')
  async initiateLogin(
    @Response({ passthrough: true }) res: ExpressResponse
  ) {
    try {
      const { authorizationUrl } =
        this.oauthService.generateAuthorizationUrl();

      this.logger.debug('Login initiated, redirecting to Google');

      return res.redirect(authorizationUrl);
    } catch (error) {
      this.logger.error('Failed to initiate login', error);
      throw new BadRequestException('Failed to initiate login');
    }
  }

  /**
   * GET /api/v1/auth/callback
   *
   * Handle OAuth callback from Google
   *
   * Query params: code, state (from Google)
   *
   * @param code - Authorization code from Google
   * @param state - State parameter for CSRF protection
   * @param res - Express response
   * @returns Redirect to dashboard with JWT cookie
   *
   * NOTE: In production, this needs database integration.
   * It will need PrismaService to:
   * 1. Lookup user by email
   * 2. Create AuthToken record
   * 3. Update User.lastLoginAt
   * 4. Log audit event
   */
  @Get('callback')
  async handleCallback(
    @Query('code') code: string,
    @Query('state') state: string,
    @Response({ passthrough: true }) res: ExpressResponse
  ) {
    try {
      // Validate input
      if (!code || !state) {
        throw new BadRequestException('Missing code or state parameter');
      }

      // TODO: In production, integrate with PrismaService
      // const loginResponse = await this.oauthService.handleCallback(
      //   { code, state },
      //   async (email) => {
      //     return prisma.user.findUnique({ where: { email } });
      //   }
      // );

      this.logger.warn(
        'Auth callback requires database integration (not implemented in Phase 1.7 reference)'
      );

      throw new BadRequestException(
        'OAuth callback handler not fully implemented. See auth.controller.ts for integration details.'
      );
    } catch (error) {
      this.logger.error('OAuth callback failed', error);

      // Redirect to login with error
      return res.redirect('/login?error=oauth_failed');
    }
  }

  /**
   * GET /api/v1/auth/me
   *
   * Get current authenticated user
   *
   * @param req - Express request (with user attached by AuthGuard)
   * @returns Current user profile
   */
  @Get('me')
  @UseGuards(AuthGuard)
  async getCurrentUser(@Request() req: any) {
    const user = req.user;

    if (!user) {
      throw new UnauthorizedException('User not found in request');
    }

    this.logger.debug(`User info requested for: ${user.id}`);

    return {
      id: user.id,
      email: user.email,
      role: user.role,
      organizationId: user.organizationId,
      accountStatus: user.accountStatus,
    };
  }

  /**
   * POST /api/v1/auth/logout
   *
   * Invalidate current session (revoke JWT)
   *
   * @param req - Express request
   * @param res - Express response
   * @returns Success response
   *
   * NOTE: In production, this needs database integration to:
   * 1. Extract JWT from request
   * 2. Mark token as revoked in AuthToken table
   * 3. Log audit event
   * 4. Clear cookie on frontend
   */
  @Post('logout')
  @UseGuards(AuthGuard)
  async logout(@Request() req: any, @Response({ passthrough: true }) res: ExpressResponse) {
    const user = req.user;

    this.logger.log(`User ${user?.id} logging out`);

    // TODO: In production, integrate with PrismaService
    // 1. Find JWT token in request
    // 2. Update AuthToken.revokedAt
    // 3. Log audit event

    // Clear JWT cookie
    res.clearCookie('jwt', {
      httpOnly: true,
      secure: false, // true in production
      sameSite: 'lax',
      path: '/',
    });

    return { status: 'logged_out' };
  }

  /**
   * POST /api/v1/auth/verify
   *
   * Verify token validity (internal use or frontend check)
   *
   * @param req - Express request
   * @returns Token validity info
   */
  @Post('verify')
  @UseGuards(AuthGuard)
  async verifyToken(@Request() req: any) {
    const user = req.user;

    if (!user) {
      throw new UnauthorizedException('Invalid token');
    }

    // Get token from request
    const token =
      req.cookies?.jwt ||
      req.headers.authorization?.substring(7);

    if (!token) {
      throw new UnauthorizedException('Token not found');
    }

    // Check expiration
    const expiresAt = this.jwtService.getExpirationDate(token);
    const ttl = this.jwtService.getTokenTTL(token);

    return {
      valid: true,
      userId: user.id,
      email: user.email,
      role: user.role,
      expiresAt: expiresAt ? Math.floor(expiresAt.getTime() / 1000) : null,
      ttl, // seconds until expiration
    };
  }
}

The `@verifit/auth` module provides complete authentication and authorization for VeriFit. This guide shows how to integrate it into your NestJS application.

// apps/api/src/app.module.ts
import { Module } from '@nestjs/common';
import { ConfigModule } from '@nestjs/config';
import { AuthModule } from '@verifit/auth';

@Module({
  imports: [
    ConfigModule.forRoot(),
    AuthModule.register({
      jwt: {
        secret: process.env.JWT_SECRET,
        expiresIn: process.env.JWT_EXPIRATION || '15m',
      },
      oauth: {
        clientId: process.env.OAUTH_CLIENT_ID,
        clientSecret: process.env.OAUTH_CLIENT_SECRET,
        redirectUri: process.env.OAUTH_REDIRECT_URI,
      },
      isGlobal: true,
    }),
  ],
})
export class AppModule {}

  }

Example 4: Full Authorization Check
typescript
import { Controller, Get, Inject, Request } from '@nestjs/common';
import { AuthorizationService, ResourceType, ResourceAction } from '@verifit/auth';

@Controller('students')
export class StudentController {
  constructor(private authz: AuthorizationService) {}

  @Get(':id')
  async getStudent(@Request() req, @Param('id') studentId: string) {
    // Manual authorization check
    this.authz.assertAuthorized({
      user: req.user,
      resource: ResourceType.STUDENT,
      action: ResourceAction.READ_STUDENT,
      resourceId: studentId,
      resourceOrgId: req.user.organizationId,
    });

    // If we reach here, user is authorized
    return { student: { id: studentId } };
  }
}
Services
JwtService
Generate and validate JWT tokens:

typescript
import { JwtService } from '@verifit/auth';

@Injectable()
export class MyService {
  constructor(private jwt: JwtService) {}

  generateToken(user) {
    return this.jwt.generateToken({
      userId: user.id,
      email: user.email,
      organizationId: user.organizationId,
      role: user.role,
    });
  }

  validateToken(token: string) {
    return this.jwt.verifyToken(token);
  }
}
