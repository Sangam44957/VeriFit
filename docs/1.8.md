/**
 * Auth Service - Business Logic for Authentication
 *
 * Responsibilities:
 * - Handle OAuth callback (user creation/lookup)
 * - Generate and store JWT tokens
 * - Revoke tokens on logout
 * - Track login attempts and account lockout
 * - Log audit events
 * - Manage OAuth connections
 *
 * Integrates:
 * - @verifit/auth (JwtService, OAuthService)
 * - PrismaService (database)
 * - AuditService (logging)
 */

import {
  Injectable,
  BadRequestException,
  UnauthorizedException,
  Logger,
} from '@nestjs/common';
import { JwtService, OAuthService, AuthenticatedUser } from '@verifit/auth';
import { PrismaService } from '@verifit/database';
import { AuditService } from '../audit/audit.service';

/**
 * Auth Service
 *
 * @example
 * ```typescript
 * const loginResponse = await authService.handleOAuthCallback(
 *   { code: '...', state: '...' },
 *   request.ip,
 *   request.headers['user-agent']
 * );
 *
 * // Set JWT cookie
 * response.cookie('jwt', loginResponse.token, {
 *   httpOnly: true,
 *   secure: true,
 *   sameSite: 'lax'
 * });
 * ```
 */
@Injectable()
export class AuthService {
  private readonly logger = new Logger(AuthService.name);

  constructor(
    private readonly jwtService: JwtService,
    private readonly oauthService: OAuthService,
    private readonly prisma: PrismaService,
    private readonly audit: AuditService
  ) {}

  /**
   * Handle OAuth callback
   *
   * Flow:
   * 1. Verify OAuth state (CSRF protection)
   * 2. Exchange code for Google access token
   * 3. Retrieve user profile
   * 4. Lookup user in database (pre-created by admin)
   * 5. Create or update OAuth connection
   * 6. Generate JWT token
   * 7. Store token in database
   * 8. Update user last login
   * 9. Log audit event
   *
   * @param request - OAuth callback request (code + state)
   * @param ipAddress - Client IP for audit
   * @param userAgent - Client user agent for audit
   * @returns Login response with JWT + user info
   * @throws BadRequestException if any step fails
   */
  async handleOAuthCallback(
    request: { code: string; state: string },
    ipAddress?: string,
    userAgent?: string
  ) {
    try {
      // Step 1-3: OAuth callback (handled by OAuthService)
      // Step 4: Lookup user in database
      const loginResponse = await this.oauthService.handleCallback(
        request,
        async (email) => {
          const user = await this.prisma.user.findUnique({
            where: { email },
          });

          if (!user) {
            this.logger.warn(
              `User not found for email: ${email}. User must be pre-created.`
            );
            return null;
          }

          return {
            id: user.id,
            email: user.email,
            role: user.role as 'ADMIN' | 'STAFF' | 'STUDENT',
            organizationId: user.organizationId,
            accountStatus: user.accountStatus as
              | 'ACTIVE'
              | 'LOCKED'
              | 'SUSPENDED',
          };
        }
      );

      const user = loginResponse.user;

      // Step 5: Create/update OAuth connection
      await this.prisma.oauthConnection.upsert({
        where: {
          provider_providerUserId: {
            provider: 'google',
            providerUserId: '', // Would come from Google profile
          },
        },
        create: {
          userId: user.id,
          provider: 'google',
          providerUserId: '', // TODO: extract from Google profile
          providerEmail: user.email,
          lastUsedAt: new Date(),
        },
        update: {
          lastUsedAt: new Date(),
        },
      });

      // Step 6-7: Generate JWT and store token
      const metadata = this.jwtService.createTokenRecordMetadata(
        loginResponse.token,
        user.id,
        ipAddress,
        userAgent
      );

      await this.prisma.authToken.create({
        data: metadata,
      });

      // Step 8: Update user last login
      await this.prisma.user.update({
        where: { id: user.id },
        data: {
          lastLoginAt: new Date(),
          lastLoginIpAddress: ipAddress,
          loginAttempts: 0, // Reset on successful login
          lockedUntil: null,
        },
      });

      // Step 9: Log audit event
      await this.audit.log({
        organizationId: user.organizationId,
        userId: user.id,
        action: 'LOGIN',
        resource: 'User',
        resourceId: user.id,
        status: 'SUCCESS',
        ipAddress,
        userAgent,
      });

      this.logger.log(`User ${user.id} logged in successfully`);

      return loginResponse;
    } catch (error) {
      // Log failed attempt
      if (error instanceof BadRequestException) {
        await this.audit.log({
          action: 'LOGIN_FAILURE',
          resource: 'User',
          status: 'FAILURE',
          ipAddress,
          userAgent,
          details: { error: error.message },
        });
      }

      throw error;
    }
  }

  /**
   * Logout user (revoke JWT)
   *
   * Flow:
   * 1. Extract JWT from request
   * 2. Revoke token in database
   * 3. Log audit event
   *
   * @param token - JWT token to revoke
   * @param userId - User ID (from JWT)
   * @param ipAddress - Client IP for audit
   * @returns Success response
   */
  async logout(token: string, userId: string, ipAddress?: string) {
    try {
      // Revoke token
      await this.prisma.authToken.update({
        where: { token },
        data: { revokedAt: new Date() },
      });

      // Log audit event
      await this.audit.log({
        userId,
        action: 'LOGOUT',
        resource: 'User',
        resourceId: userId,
        status: 'SUCCESS',
        ipAddress,
      });

      this.logger.log(`User ${userId} logged out`);

      return { status: 'logged_out' };
    } catch (error) {
      this.logger.error(`Logout failed for user ${userId}`, error);
      throw new BadRequestException('Logout failed');
    }
  }

  /**
   * Track failed login attempt
   *
   * Flow:
   * 1. Increment loginAttempts
   * 2. Lock account after 5 attempts (15 min)
   * 3. Log audit event
   *
   * @param email - User email
   * @param ipAddress - Client IP for audit
   */
  async trackFailedLoginAttempt(email: string, ipAddress?: string) {
    try {
      const user = await this.prisma.user.findUnique({
        where: { email },
      });

      if (!user) {
        return; // Don't reveal if user exists
      }

      const attempts = user.loginAttempts + 1;

      let accountStatus = user.accountStatus;
      let lockedUntil: Date | null = null;

      // Lock account after 5 failed attempts
      if (attempts >= 5) {
        accountStatus = 'LOCKED';
        lockedUntil = new Date(Date.now() + 15 * 60 * 1000); // 15 min
      }

      await this.prisma.user.update({
        where: { id: user.id },
        data: {
          loginAttempts: attempts,
          accountStatus,
          lockedUntil,
        },
      });

      // Log attempt
      await this.audit.log({
        userId: user.id,
        organizationId: user.organizationId,
        action: 'LOGIN_FAILURE',
        resource: 'User',
        resourceId: user.id,
        status: 'FAILURE',
        ipAddress,
        details: {
          attempt: attempts,
          accountLocked: accountStatus === 'LOCKED',
        },
      });

      if (accountStatus === 'LOCKED') {
        this.logger.warn(
          `Account locked for ${email} after 5 failed attempts`
        );
      }
    } catch (error) {
      this.logger.error(`Failed to track login attempt for ${email}`, error);
    }
  }

  /**
   * Verify token and return user
   *
   * Flow:
   * 1. Validate JWT
   * 2. Check revocation status in database
   * 3. Return user if valid, null if invalid
   *
   * @param token - JWT token
   * @returns User if valid, null if invalid/revoked
   */
  async verifyTokenAndGetUser(token: string): Promise<AuthenticatedUser | null> {
    try {
      // Validate JWT
      const payload = this.jwtService.verifyToken(token);

      if (!payload) {
        return null;
      }

      // Check revocation status
      const authToken = await this.prisma.authToken.findUnique({
        where: { token },
      });

      if (!authToken || authToken.revokedAt !== null) {
        this.logger.warn(
          `Token revoked or not found for user ${payload.userId}`
        );
        return null;
      }

      // Token is valid
      return {
        id: payload.userId,
        email: payload.email,
        organizationId: payload.organizationId,
        role: payload.role,
        accountStatus: 'ACTIVE',
      };
    } catch (error) {
      this.logger.debug(`Token verification failed: ${error}`);
      return null;
    }
  }

  /**
   * Create test JWT for development
   *
   * Development only: Create a valid JWT for testing
   *
   * @param userId - User ID
   * @param email - User email
   * @param organizationId - Organization ID
   * @param role - User role
   * @returns JWT token (not stored in database)
   */
  async createTestToken(
    userId: string,
    email: string,
    organizationId: string,
    role: 'ADMIN' | 'STAFF' | 'STUDENT'
  ): Promise<string> {
    this.logger.warn(
      `Creating test token for user ${userId} - should not be used in production`
    );

    return this.jwtService.generateToken({
      userId,
      email,
      organizationId,
      role,
    });
  }
}

/**
 * Audit Service - Event Logging
 *
 * Logs all authentication and authorization events
 * to audit trail for compliance and debugging.
 *
 * Writes asynchronously via BullMQ (non-blocking).
 */

import { Injectable, Logger } from '@nestjs/common';
import { PrismaService } from '@verifit/database';

export interface AuditLogInput {
  organizationId?: string;
  userId?: string;
  action: string;
  resource: string;
  resourceId?: string;
  status: 'SUCCESS' | 'FAILURE' | 'DENIED';
  ipAddress?: string;
  userAgent?: string;
  details?: Record<string, any>;
}

/**
 * Audit Service - logs events asynchronously
 */
@Injectable()
export class AuditService {
  private readonly logger = new Logger(AuditService.name);

  constructor(private readonly prisma: PrismaService) {}

  /**
   * Log audit event
   *
   * Writes to database (synchronously for now).
   * Phase 6: Move to BullMQ for async processing.
   *
   * @param input - Audit log input
   */
  async log(input: AuditLogInput): Promise<void> {
    try {
      await this.prisma.auditEvent.create({
        data: {
          organizationId: input.organizationId || 'unknown',
          userId: input.userId,
          action: input.action,
          resource: input.resource,
          resourceId: input.resourceId,
          status: input.status,
          ipAddress: input.ipAddress,
          userAgent: input.userAgent,
          details: input.details,
        },
      });
    } catch (error) {
      // Log error but don't throw (audit failure shouldn't break request)
      this.logger.error(`Failed to write audit log: ${error}`);
    }
  }

  /**
   * Get audit events for organization
   *
   * @param organizationId - Organization ID
   * @param limit - Number of events to return
   * @returns List of audit events
   */
  async getEvents(organizationId: string, limit: number = 100) {
    return this.prisma.auditEvent.findMany({
      where: { organizationId },
      take: limit,
      orderBy: { createdAt: 'desc' },
      include: { user: true },
    });
  }
}

/**
 * Auth Controller - Authentication Endpoints
 *
 * Endpoints:
 * GET    /api/v1/auth/login         - Initiate OAuth login
 * GET    /api/v1/auth/callback      - Handle OAuth callback
 * POST   /api/v1/auth/logout        - Logout (revoke token)
 * GET    /api/v1/auth/me            - Get current user
 * POST   /api/v1/auth/verify        - Verify token validity
 */

import {
  Controller,
  Get,
  Post,
  Query,
  Request,
  Response,
  UseGuards,
  BadRequestException,
  Logger,
} from '@nestjs/common';
import { Response as ExpressResponse } from 'express';
import { AuthGuard } from '@verifit/auth';
import { AuthService } from './auth.service';
import { z } from 'zod';

/**
 * Validation schemas
 */
const OAuthCallbackSchema = z.object({
  code: z.string().min(1, 'Authorization code required'),
  state: z.string().min(1, 'State parameter required'),
});

/**
 * Auth Controller
 *
 * Handles OAuth login flow and token validation.
 */
@Controller('api/v1/auth')
export class AuthController {
  private readonly logger = new Logger(AuthController.name);

  constructor(private readonly authService: AuthService) {}

  /**
   * GET /api/v1/auth/login
   *
   * Initiate OAuth login (redirect to Google)
   *
   * @param res - Express response (for redirect)
   * @returns Redirect to Google OAuth consent screen
   *
   * @example
   * ```
   * GET /api/v1/auth/login
   * 302 Redirect to https://accounts.google.com/o/oauth2/v2/auth?...
   * ```
   */
  @Get('login')
  async initiateLogin(
    @Response({ passthrough: true }) res: ExpressResponse
  ) {
    try {
      const { authorizationUrl } = this.authService['oauthService']
        ? this.authService['oauthService'].generateAuthorizationUrl()
        : { authorizationUrl: '' };

      if (!authorizationUrl) {
        throw new BadRequestException(
          'Failed to generate authorization URL'
        );
      }

      this.logger.debug('Login initiated, redirecting to Google');

      return res.redirect(authorizationUrl);
    } catch (error) {
      this.logger.error('Failed to initiate login', error);
      throw new BadRequestException('Failed to initiate login');
    }
  }

  /**
   * GET /api/v1/auth/callback
   *
   * Handle OAuth callback from Google
   *
   * Query params: code, state (from Google)
   *
   * Flow:
   * 1. Validate code and state
   * 2. Exchange code for access token (handled by OAuthService)
   * 3. Create/update user and OAuth connection
   * 4. Generate JWT and store in database
   * 5. Set JWT cookie
   * 6. Redirect to dashboard
   *
   * @param code - Authorization code from Google
   * @param state - State parameter for CSRF protection
   * @param res - Express response
   * @returns Redirect to dashboard with JWT cookie
   *
   * @example
   * ```
   * GET /api/v1/auth/callback?code=...&state=...
   * 302 Redirect to /dashboard (with jwt cookie set)
   * ```
   */
  @Get('callback')
  async handleCallback(
    @Query('code') code: string,
    @Query('state') state: string,
    @Request() req: any,
    @Response({ passthrough: true }) res: ExpressResponse
  ) {
    try {
      // Validate input
      OAuthCallbackSchema.parse({ code, state });

      // Handle OAuth callback
      const loginResponse = await this.authService.handleOAuthCallback(
        { code, state },
        req.ip,
        req.headers['user-agent']
      );

      // Set JWT cookie (HttpOnly, Secure, SameSite)
      res.cookie('jwt', loginResponse.token, {
        httpOnly: true,
        secure: process.env.NODE_ENV === 'production', // true in prod
        sameSite: 'lax',
        maxAge: 7 * 24 * 60 * 60 * 1000, // 7 days
        path: '/',
      });

      this.logger.log(
        `User ${loginResponse.user.id} logged in via OAuth`
      );

      // Redirect to dashboard
      return res.redirect(
        process.env.FRONTEND_URL || 'http://localhost:3000/dashboard'
      );
    } catch (error) {
      this.logger.error('OAuth callback failed', error);

      // Redirect to login with error
      const errorParam = error instanceof Error ? error.message : 'unknown';
      return res.redirect(
        `${process.env.FRONTEND_URL || 'http://localhost:3000'}/login?error=${encodeURIComponent(errorParam)}`
      );
    }
  }

  /**
   * GET /api/v1/auth/me
   *
   * Get current authenticated user
   *
   * @param req - Express request (with user attached by AuthGuard)
   * @returns Current user profile
   *
   * @example
   * ```
   * GET /api/v1/auth/me
   * Authorization: Bearer <jwt>
   *
   * 200 OK
   * {
   *   "id": "user-123",
   *   "email": "student@lpu.edu",
   *   "role": "STUDENT",
   *   "organizationId": "org-456",
   *   "accountStatus": "ACTIVE"
   * }
   * ```
   */
  @Get('me')
  @UseGuards(AuthGuard)
  async getCurrentUser(@Request() req: any) {
    const user = req.user;

    if (!user) {
      throw new BadRequestException('User not found in request');
    }

    this.logger.debug(`User info requested for: ${user.id}`);

    return {
      id: user.id,
      email: user.email,
      role: user.role,
      organizationId: user.organizationId,
      accountStatus: user.accountStatus,
    };
  }

  /**
   * POST /api/v1/auth/logout
   *
   * Invalidate current session (revoke JWT)
   *
   * Flow:
   * 1. Extract JWT from cookie/header
   * 2. Revoke token in database
   * 3. Clear JWT cookie
   * 4. Log audit event
   *
   * @param req - Express request
   * @param res - Express response
   * @returns Success response
   *
   * @example
   * ```
   * POST /api/v1/auth/logout
   * Authorization: Bearer <jwt>
   *
   * 200 OK
   * { "status": "logged_out" }
   * ```
   */
  @Post('logout')
  @UseGuards(AuthGuard)
  async logout(
    @Request() req: any,
    @Response({ passthrough: true }) res: ExpressResponse
  ) {
    const user = req.user;

    if (!user) {
      throw new BadRequestException('User not found');
    }

    // Extract JWT from cookie or header
    const token =
      req.cookies?.jwt ||
      req.headers.authorization?.substring(7);

    if (!token) {
      throw new BadRequestException('Token not found');
    }

    // Revoke token
    await this.authService.logout(token, user.id, req.ip);

    // Clear JWT cookie
    res.clearCookie('jwt', {
      httpOnly: true,
      secure: process.env.NODE_ENV === 'production',
      sameSite: 'lax',
      path: '/',
    });

    this.logger.log(`User ${user.id} logged out`);

    return { status: 'logged_out' };
  }

  /**
   * POST /api/v1/auth/verify
   *
   * Verify token validity (internal use or frontend check)
   *
   * @param req - Express request
   * @returns Token validity info
   *
   * @example
   * ```
   * POST /api/v1/auth/verify
   * Authorization: Bearer <jwt>
   *
   * 200 OK
   * {
   *   "valid": true,
   *   "userId": "user-123",
   *   "email": "student@lpu.edu",
   *   "role": "STUDENT",
   *   "expiresAt": 1234567890,
   *   "ttl": 3600
   * }
   * ```
   */
  @Post('verify')
  @UseGuards(AuthGuard)
  async verifyToken(@Request() req: any) {
    const user = req.user;

    if (!user) {
      throw new BadRequestException('Invalid token');
    }

    // Extract token
    const token =
      req.cookies?.jwt ||
      req.headers.authorization?.substring(7);

    if (!token) {
      throw new BadRequestException('Token not found');
    }

    // Get expiration info
    const expiresAt = (this.authService as any).jwtService.getExpirationDate(
      token
    );
    const ttl = (this.authService as any).jwtService.getTokenTTL(token);

    return {
      valid: true,
      userId: user.id,
      email: user.email,
      role: user.role,
      expiresAt: expiresAt ? Math.floor(expiresAt.getTime() / 1000) : null,
      ttl,
    };
  }
}
