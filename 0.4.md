5. The @/* alias is unsafe for the worker runtime
The task uses:

and:

  '@': path.resolve(__dirname, './src'),
}
This creates a runtime problem.

TypeScript can understand an alias, but after:

tsc

Node does not magically understand that alias.

For a lightweight Node worker, we do not need to introduce a runtime alias mechanism just for this.

That gives us:

TypeScript source
      ↓
tsc
      ↓
real .js files
      ↓
Node runs them directly
Much cleaner.

This keeps:

11. Health check has an important hidden failure mode
This part looks good:

await this.redis.ping();
but the Redis connection is configured with:

maxRetriesPerRequest: null,
enableOfflineQueue: true
BullMQ recommends maxRetriesPerRequest: null for worker connections because the worker should keep retrying while Redis is unavailable. 

That is correct for the worker connection.

But it is not necessarily correct for the HTTP readiness probe.

Otherwise this can happen:

GET /health/ready
       ↓
redis.ping()
       ↓
Redis unavailable
       ↓
ioredis keeps retrying
       ↓
HTTP request waits instead of promptly returning 503
That is a bad readiness endpoint.

Better design
Use a separate short-lived/readiness Redis client, or perform a bounded connectivity check with a timeout.

Conceptually:

BullMQ Redis connection
    └── long-lived / retry forever

Health Redis connection
    └── short bounded probe
    └── failure → 503 quickly
This is important for production behavior.

12. Worker health says "active workers" but actually reports registered workers
This:

workers: {
  count: activeWorkers.length,
  active: activeWorkers,
}
is misleading.

getWorkers() returns registered Worker instances, not jobs currently active.

At this stage we have no processors anyway.

Correction
Either report:

workers: {
  registered: [...]
}
or remove worker information from readiness until actual workers exist.

I prefer the latter for Task 0.4.

13. Graceful shutdown is not actually fully graceful
The code has:

server.close();
await workerManager.shutdown();
await redis.quit();
process.exit(0);
The problem is:

server.close();
is not awaited.

So the process may exit before the HTTP server has actually completed closing existing connections.

Better
Wrap it:

await new Promise<void>((resolve, reject) => {
  server.close((error) => {
    if (error) {
      reject(error);
      return;
    }

    resolve();
  });
});
Then close worker resources and Redis in a defined order.

Also guard against duplicate SIGTERM/SIGINT handling so shutdown isn't executed twice.

20. One thing from BullMQ that we should consciously preserve

Every VeriFit job processor must assume at-least-once delivery
and therefore be idempotent or safely deduplicated.
That's an important long-term invariant.

/**
 * Redis connection configuration.
 * All workers and queues use the same Redis instance.
 */
export const RedisConfigSchema = z.object({
  host: z.string().default('localhost'),
  port: z.coerce.number().int().default(6379),
  password: z.string().optional(),
  database: z.coerce.number().int().default(0),
  maxRetries: z.coerce.number().int().default(3),
  enableReadyCheck: z.coerce.boolean().default(true),
  enableOfflineQueue: z.coerce.boolean().default(true),
});

export type RedisConfig = z.infer<typeof RedisConfigSchema>;

/**
 * Load Redis config from environment.
 */
export function loadRedisConfig(): RedisConfig {
  const config = RedisConfigSchema.parse({
    host: process.env.REDIS_HOST,
    port: process.env.REDIS_PORT,
    password: process.env.REDIS_PASSWORD,
    database: process.env.REDIS_DB,
  });

  return config;
}

/**
 * Format connection URL for debugging/logging.
 */
export function getRedisConnectionUrl(config: RedisConfig): string {
  return `redis://${config.password ? '***:' : ''}${config.host}:${config.port}/${config.database}`;
}

import { Worker, QueueBase } from 'bullmq';
import { Redis } from 'ioredis';
import { QUEUE_NAMES, QUEUE_CONFIG, type QueueName } from '@/config/queue';
import { logger } from '@/lib/logger';

/**
 * Manages all background job workers.
 * Workers are created on-demand; only active workers consume from queues.
 */
export class WorkerManager {
  private workers: Map<QueueName, Worker> = new Map();
  private redis: Redis;

  constructor(redis: Redis) {
    this.redis = redis;
  }

  /**
   * Register a worker for a specific queue.
   * The worker will not be active until started.
   */
  registerWorker(
    queueName: QueueName,
    processor: (job: any) => Promise<any>,
  ): void {
    if (this.workers.has(queueName)) {
      logger.warn(`Worker for queue "${queueName}" already registered, skipping`);
      return;
    }

    const config = QUEUE_CONFIG[queueName];
    if (!config) {
      throw new Error(`No configuration found for queue: ${queueName}`);
    }

    const worker = new Worker(queueName, processor, {
      connection: this.redis,
      concurrency: config.maxConcurrency,
      settings: {
        maxStalledCount: 2,
        maxStalledInterval: 30000,
        stalledInterval: 5000,
      },
    });

    // Event handlers
    worker.on('completed', (job) => {
      logger.info(`Job completed: ${queueName}#${job.id}`);
    });

    worker.on('failed', (job, err) => {
      logger.error(
        `Job failed: ${queueName}#${job?.id}, attempt ${job?.attemptNumber}`,
        err,
      );
    });

    worker.on('error', (err) => {
      logger.error(`Worker error on queue "${queueName}":`, err);
    });

    this.workers.set(queueName, worker);
    logger.info(`Worker registered for queue: ${queueName}`);
  }

  /**
   * Get all registered workers.
   */
  getWorkers(): Map<QueueName, Worker> {
    return this.workers;
  }

  /**
   * Get a specific worker.
   */
  getWorker(queueName: QueueName): Worker | undefined {
    return this.workers.get(queueName);
  }

  /**
   * Graceful shutdown: close all workers.
   */
  async shutdown(): Promise<void> {
    logger.info(`Shutting down ${this.workers.size} worker(s)...`);

    const promises = Array.from(this.workers.values()).map(async (worker) => {
      try {
        await worker.close();
      } catch (error) {
        logger.error('Error closing worker:', error);
      }
    });

    await Promise.all(promises);
    this.workers.clear();
    logger.info('All workers shut down');
  }
}
Explanation:

Manages all BullMQ workers
Workers created on-demand
Event handlers for success/failure
Graceful shutdown support
Important: No actual job processors yet (added in phase 2+)

import { Redis } from 'ioredis';
import { logger } from '@/lib/logger';

export interface WorkerHealth {
  status: 'healthy' | 'unhealthy';
  timestamp: string;
  uptime: number;
  redis: {
    status: 'connected' | 'disconnected';
    latency?: number;
  };
  workers: {
    count: number;
    active: string[];
  };
}

export class HealthService {
  private startTime = Date.now();

  constructor(private redis: Redis) {}

  async getHealth(activeWorkers: string[]): Promise<WorkerHealth> {
    let redisStatus: 'connected' | 'disconnected' = 'disconnected';
    let redisLatency: number | undefined;

    try {
      const start = Date.now();
      // Simple ping to Redis
      await this.redis.ping();
      redisLatency = Date.now() - start;
      redisStatus = 'connected';
    } catch (error) {
      logger.warn('Redis connection check failed:', error);
      redisStatus = 'disconnected';
    }

    const uptime = Date.now() - this.startTime;
    const status = redisStatus === 'connected' ? 'healthy' : 'unhealthy';

    return {
      status,
      timestamp: new Date().toISOString(),
      uptime,
      redis: {
        status: redisStatus,
        latency: redisLatency,
      },
      workers: {
        count: activeWorkers.length,
        active: activeWorkers,
      },
    };
  }
}

import { Redis } from 'ioredis';
import express from 'express';
import { loadRedisConfig, getRedisConnectionUrl } from '@/config/redis';
import { WorkerManager } from '@/worker/worker-manager';
import { HealthService } from '@/health/health-service';
import { logger } from '@/lib/logger';

/**
 * Main worker application.
 * - Connects to Redis
 * - Creates HTTP health check server
 * - Registers job workers (added incrementally in later phases)
 */

async function main() {
  logger.info('VeriFit Worker starting...');

  // Redis connection
  const redisConfig = loadRedisConfig();
  const redis = new Redis({
    host: redisConfig.host,
    port: redisConfig.port,
    password: redisConfig.password,
    db: redisConfig.database,
    maxRetriesPerRequest: null,
    enableReadyCheck: redisConfig.enableReadyCheck,
    enableOfflineQueue: redisConfig.enableOfflineQueue,
  });

  redis.on('connect', () => {
    logger.info(`Redis connected: ${getRedisConnectionUrl(redisConfig)}`);
  });

  redis.on('error', (err) => {
    logger.error('Redis error:', err);
  });

  // Worker manager
  const workerManager = new WorkerManager(redis);
  logger.info('WorkerManager created');

  // Health check service
  const healthService = new HealthService(redis);

  // Express server for health checks
  const app = express();
  const port = process.env.PORT || 3002;

  app.get('/health/live', (_req, res) => {
    // Liveness: process is running
    res.json({ status: 'alive' });
  });

  app.get('/health/ready', async (_req, res) => {
    // Readiness: dependencies available
    const health = await healthService.getHealth(
      Array.from(workerManager.getWorkers().keys()),
    );
    const statusCode = health.status === 'healthy' ? 200 : 503;
    res.status(statusCode).json(health);
  });

  const server = app.listen(port, () => {
    logger.info(`Worker health server listening on port ${port}`);
  });

  // Graceful shutdown
  const shutdown = async () => {
    logger.info('Shutdown signal received');
    server.close();
    await workerManager.shutdown();
    await redis.quit();
    logger.info('Worker shutdown complete');
    process.exit(0);
  };

  process.on('SIGTERM', shutdown);
  process.on('SIGINT', shutdown);

  logger.info('VeriFit Worker ready');
}

main().catch((error) => {
  logger.error('Worker startup failed:', error);
  process.exit(1);
});
